
\subsection{Outer Measure}
\begin{definition}[Open box]
    An open box $B\subseteq\mathbb{R}^n$ is any of the form 
    $$B=\prod_{i=1}^{n}(a_i, b_i)=\{(x_1,x_2,\cdots,x_n)\in\mathbb{R}^n:a_i<x_i<b_i, i=1,2,\cdots,n\}.$$
    We define the volume of $B$ to be 
    $$Vol(B)=\prod_{i=1}^{n}(b_i-a_i).$$
\end{definition}

\begin{definition}
    Let $\Omega\in\mathbb{R}^n$. We say a collection $(B_j)_{j\in\mathcal{J}}$ of boxes covering $\Omega$ if $\Omega\subseteq\bigcup_{j\in\mathcal{J}}B_j.$
\end{definition}

\begin{definition}[Outer Measure]
    If $\Omega\in\mathbb{R}^n,$ we define the outer measure of $\Omega$ to be the quantity
    \[m^*(\Omega)=\inf\left\{\sum_{j\in\mathcal{J}}Vol(B_j): (B_j)_{j\in\mathcal{J}} \text{ is a countable collection of boxes covering } \Omega\right\}.\]
\end{definition}

\begin{remark}
    $0\le m^*(\Omega)\le\infty.$
\end{remark}

\begin{theorem}\ \vspace{-2em}
    \begin{enumerate}
        \item (Empty set) $m^*(\varnothing)=0.$
        \item (Monotonicity) If $A\subseteq B\subseteq\mathbb{R}^n$, then $m^*(A)\le m^*(B).$
        \item (Countable sub-additivity) For a countable collection $(A_j)_{j\in\mathcal{J}}$ in $\mathbb{R}^n$, we have $m^*(\cup_{j\in\mathcal{J}}A_j)<\sum_{j\in\mathcal{J}}m^*(A_j).$ 
        \item (Translation Invariance) For $\Omega\subseteq\mathbb{R}^n$ and $x\in\mathbb{R}^n$, we have $m^*(\Omega+x)=m^*(\Omega).$
    \end{enumerate}
\end{theorem}

\begin{pfbox}
    For 1, given any $\epsilon>0,$ $(0,\epsilon^{\frac{1}{n}})^{n}$ is an open boxes covering $\varnothing$.
    By definition, $0\le m^*(\varnothing)\le Vol((0,\epsilon^{\frac{1}{n}})^{n})=\epsilon.$ So $m^*(\Omega)=0.$\\

    For 2, if $m^*(B)=\infty,$ there is nothing to prove. Now assume $m^*(B)<\epsilon,$ given any $\epsilon>0$,
    there exists a countable collection of open boxes $(B_j)_{j\in\mathcal{J}}$ of $B$ s.t. $\sum_{j\in\mathcal{J}}Vol(B_j)\le m^*(B)+\epsilon$.
    Since $A\subseteq B\subseteq \cup_{j\in\mathcal{J}}B_j$, $m^*(A)\le \sum_{j\in\mathcal{J}}Vol(B_j)\le m^*(B)+\epsilon.$ Take $\epsilon\to 0,$ we have $m^*(A)\le m^*(B).$\\

    For 3, if $m^*(A_j)=\infty$ for some $j\in\mathcal{J},$ then we are done. Now assume $m^*(A_j)<\infty\,\forall j\in\mathcal{J}.$ Given any $\epsilon>0,$ there exist a countable collection of open boxes $\{B_{j,k}\}_{k\in\mathbb{N}}$ of $B_j$ s.t. $\sum_{k=1}^{\infty}Vol(B_j,k)\le m^*(A_j)+\frac{\epsilon}{2^j}$.
    Since $$\bigcup_{j\in\mathbb{N}} A_j\subseteq \bigcup_{j\in\mathbb{N}}\bigcup_{k\in\mathbb{N}} B_{j,k},\quad m^*(\bigcup_{j\in\mathbb{N}}A_j)\le\sum_{{j\in\mathbb{N}}}\sum_{{k\in\mathbb{N}}} Vol(B_{j,k})\le \sum_{j\in\mathbb{N}}m^*(A_j)+\epsilon.$$
    Take $\epsilon\to 0,$ then we have 
    $$m^*(\bigcup_{j\in\mathcal{J}}A_j)<\sum_{j\in\mathcal{J}}m^*(A_j).$$\\

    For 4, since $\sum_{j\in\mathbb{N}}Vol(B_j) = \sum_{j\in\mathbb{N}}Vol(B_j+x),$ by the definition of outer measure, we have $m^*(x+\Omega)=m^*(\Omega).$
\end{pfbox}



\begin{proposition}
    Let $B=\prod_{i=1}^{n}[a_i, b_i]=\left\{ (x_1,x_2,\cdots,x_n)\in\mathbb{R}^n:a_i\le x_i\le b_i, i=1,2,\cdots,n\right\},$ we have $m^*(B)=\prod_{i=1}^{n}(b_i-a_i).$
\end{proposition}

\begin{pfbox}
    $(\le)$ Let an open box $U_\epsilon = \prod_{i=1}^{n}(a_i+\epsilon, b_i-\epsilon)$ covering $B$. By definition, $m^*(B)\le Vol(U_\epsilon)=\prod_{i=1}^{n}(b_i-a_i+2\epsilon).$ Take $\epsilon\to 0$, then we have $m^*(B)\le \prod_{i=1}^{n}(b_i-a_i).$\\
    
    $(\ge)$ Let $Vol(B)=\prod_{i=1}^{n}(b_i-a_i)$ and any countable collection of open boxes $\{B_j\}_{j\in\mathcal{J}}$ covering B. If we can show $Vol(B)\le\sum_{j\in\mathcal{J}}Vol(B_j),$ then $\prod_{i=1}^{n}(b_i-a_i)\le m^*(B).$
    Since $B$ is compact, there exists an finite subcover of open boxes of $\{B_j\}_{j\in\mathcal{J}}$, denoted by $\{U_j\}_{j=1}^{k}$ s.t.
    $B\subseteq\cup_{j=1}^{k}U_j$ and $\sum_{j=1}^{k}Vol(U_j)\le\sum_{j=1}^{\infty}Vol(B_j)$.
    Choose a closed box $Q$ containing $B$ and $U_j$ for $j=1,2,\cdots,k.$
    Define a function $f$ on $Q$ s.t.
    \[\begin{aligned}
        f(x)=\begin{cases}
            1 \text{ if } x\in B\\
            0 \text{ if } x\in Q\setminus B,
        \end{cases}\qquad
        f_j(x)=\begin{cases}
            1 \text{ if } x\in U_j\\
            0 \text{ if } x\in Q\setminus U_j,
        \end{cases}
    \end{aligned}\]

    where $Vol(B)=\int_Q f(x)dx$ and $Vol(U_j)=\int_Qf_j(x)dx$.
    If $x\in B,\, f(x)=1,$ and there exists an integer $\mathscr{l}$ s.t. $f_\mathscr{l}(x)=1.$
    If $x\in Q\setminus B,\,f(x)=0.$ 
    So we have $f(x)\le\sum_{j=1}^{k}f_j(x).$ By the comparison property of Riemann integral,
    $\int_Qf(x)dx\le\int_Q\sum_{j=1}^{k}f_j(x)dx = \sum_{j=1}^{k}\int_Q f_j(x)dx,$ which means
    $Vol(B)\le \sum_{j=1}^{k}Vol(B_j)\le \sum_{j=1}^{\infty}Vol(B_j)$.
    Hence $Vol(B)= \prod_{i=1}^{n}(b_i-a_i)\le m^*(B).$
\end{pfbox}

\begin{example}
Consider the $m^*_1$ measure on $\mathbb{R}$.
\begin{enumerate}
    \item $m^*(\mathbb{R})=\infty.$
    \item $m^*(\mathbb{Q})=0.$\\
    We know $\mathbb{Q}$ is countable. Let $\mathbb{Q}=\cup_{q\in\mathbb{Q}}\{q\},\,m^*(\mathbb{Q})\le\sum_{q\in\mathbb{Q}}m^*(\{q\})=0.$
    \item $m^*(\mathbb{R}\setminus\mathbb{Q})=\infty.$\\
    Observe that $\mathbb{R}=(\mathbb{R}\setminus\mathbb{Q})\bigcup(\mathbb{Q})$,
    so $m^*(\mathbb{R})\le m^*(\mathbb{R}\setminus\mathbb{Q})+m^*(\mathbb{Q})$, by 1 and 2, we have $m^*(\mathbb{R}\mathbb{Q})=\infty.$ 
    This shows that $\mathbb{R}\setminus\mathbb{Q}$ is uncountable, since the set of irrational numbers has measure nonzero.
\end{enumerate}
\end{example}

\begin{remark}
    Every countable set has measure zero.
\end{remark}


\begin{lemma}[Two Properties of Cosets]
    For any $x, y \in \mathbb{R}$, let $x+\mathbb{Q} = \{x+q : q \in \mathbb{Q}\}$ be a coset of $\mathbb{Q}$ in $\mathbb{R}$. The following two properties hold:
    \begin{enumerate}
        \item If $(x+\mathbb{Q}) \cap (y+\mathbb{Q}) \neq \varnothing$, then $x+\mathbb{Q} = y+\mathbb{Q}$.
        \item Every coset meets the interval $[0,1]$, i.e., $(x+\mathbb{Q}) \cap [0,1] \neq \varnothing$.
    \end{enumerate}
\end{lemma}

\begin{pfbox}
    \textbf{Proof of 1 (Disjoint or Identical):}
    Suppose $(x+\mathbb{Q}) \cap (y+\mathbb{Q}) \neq \varnothing$. Let $z$ be an element in this intersection. 
    Then $z = x+q_1$ and $z = y+q_2$ for some $q_1, q_2 \in \mathbb{Q}$. 
    This implies $x - y = q_2 - q_1$. Since the rationals are closed under subtraction, $x - y \in \mathbb{Q}$.
    Now, for any element $a \in x+\mathbb{Q}$, we can write $a = x+r$ for some $r \in \mathbb{Q}$. 
    Substituting $x = y + (q_2 - q_1)$, we have:
    $$ a = y + (q_2 - q_1) + r = y + (q_2 - q_1 + r). $$
    Since $q_2 - q_1 + r \in \mathbb{Q}$, we get $a \in y+\mathbb{Q}$. This shows $x+\mathbb{Q} \subseteq y+\mathbb{Q}$. 
    By symmetry, $y+\mathbb{Q} \subseteq x+\mathbb{Q}$, so the two cosets are exactly equal.

    \textbf{Proof of 2 (Intersection with $[0,1]$):}
    Let $x+\mathbb{Q}$ be any coset. We want to find a rational number $q$ such that $x+q \in [0,1]$.
    This is equivalent to requiring $0 \le x+q \le 1$, or simply:
    $$ -x \le q \le 1-x. $$
    Because the set of rational numbers $\mathbb{Q}$ is dense in $\mathbb{R}$, any interval with positive length contains a rational number. 
    Since the interval $[-x, 1-x]$ has length $1 > 0$, it must contain at least one rational number $q$. 
    For such $q$, we have $x+q \in [0,1]$. Therefore, $(x+\mathbb{Q}) \cap [0,1] \neq \varnothing$.
\end{pfbox}

\begin{remark}
    The properties of these cosets stem directly from the fact that they are equivalence classes. 
    Define a relation $\sim$ on $\mathbb{R}$ by:
    $$ x \sim y \iff x-y \in \mathbb{Q}. $$
    This is an equivalence relation because it satisfies:
    \begin{itemize}
        \item \textbf{Reflexivity:} $x-x = 0 \in \mathbb{Q} \implies x \sim x$.
        \item \textbf{Symmetry:} $x \sim y \implies x-y \in \mathbb{Q} \implies -(x-y) = y-x \in \mathbb{Q} \implies y \sim x$.
        \item \textbf{Transitivity:} $x \sim y$ and $y \sim z \implies x-y \in \mathbb{Q}$ and $y-z \in \mathbb{Q} \implies (x-y)+(y-z) = x-z \in \mathbb{Q} \implies x \sim z$.
    \end{itemize}
    The equivalence class of an element $x \in \mathbb{R}$ under this relation is exactly its coset:
    $$ [x] = \{y \in \mathbb{R} : y \sim x\} = \{y \in \mathbb{R} : y-x \in \mathbb{Q}\} = \{x+q : q \in \mathbb{Q}\} = x+\mathbb{Q}. $$
    A fundamental theorem of equivalence relations states that equivalence classes form a \textbf{partition} of the underlying set ($\mathbb{R}$). This means that any two equivalence classes (cosets) are either completely disjoint or exactly identical, which is precisely the underlying mechanism of \textbf{Property 1}.
\end{remark}



\begin{proposition}[Failure of Countable Additivity]
    There exists a countable collection $(A_j)_{j \in J}$ of disjoint subsets of $\mathbb{R}$ such that
    \[ m^*\left(\bigcup_{j \in J}A_j\right) \neq \sum_{j \in J}m^*(A_j). \]
\end{proposition}

\begin{pfbox}
    We construct such a family from the cosets of $\mathbb{Q}$ in $\mathbb{R}$.
    
    \textbf{Step 1 (Choose one representative):} Let $\mathbb{R}/\mathbb{Q}$ denote the collection of all cosets. By Property 2 of our previous Lemma, we know that $A \cap [0,1] \neq \varnothing$ for every coset $A$. Thus, by the Axiom of Choice, we can choose exactly one representative $x_A \in A \cap [0,1]$ for each coset $A \in \mathbb{R}/\mathbb{Q}$. 
    Define $E = \{x_A : A \in \mathbb{R}/\mathbb{Q}\}$. By construction, $E \subseteq [0,1]$.
    
    \textbf{Step 2 (Rational translates of $E$):} Consider the set $X = \bigcup_{q \in \mathbb{Q} \cap [-1,1]} (q+E)$.
    \textit{Claim:} $[0,1] \subseteq X \subseteq [-1,2]$.
    Let $y \in [0,1]$. There exists a coset $A$ such that $y \in A$. Since $x_A \in A$ is the representative of the same coset, we have $y - x_A = q$ for some $q \in \mathbb{Q}$. 
    Since $y \in [0,1]$ and $x_A \in [0,1]$, we get $q = y - x_A \in [-1,1]$. 
    Thus, $y = q + x_A \in q+E \subseteq X$. So $[0,1] \subseteq X$.
    Also, since $E \subseteq [0,1]$ and $q \in [-1,1]$, $q+E \subseteq [-1,2]$. Thus $X \subseteq [-1,2]$.
    
    \textbf{Step 3 (Disjointness):} The collection $\{q+E : q \in \mathbb{Q} \cap [-1,1]\}$ is pairwise disjoint.
    Suppose by contradiction that $(q+E) \cap (q'+E) \neq \varnothing$ for $q \neq q'$. Then $q+x_A = q'+x_{A'}$ for some $x_A, x_{A'} \in E$. 
    This implies $x_A - x_{A'} = q' - q \in \mathbb{Q}$. By Property 1 of our Lemma, since their difference is rational, $x_A$ and $x_{A'}$ belong to the same coset. 
    However, our construction in Step 1 chose exactly one representative per coset, so we must have $x_A = x_{A'}$. This forces $q = q'$, which is a contradiction.
    
    \textbf{Step 4 (Measure contradiction):} By Step 2, $m^*([0,1]) \le m^*(X) \le m^*([-1,2])$, which implies $1 \le m^*(X) \le 3$.
    Assume countable additivity holds. Since $m^*$ is translation-invariant, $m^*(q+E) = m^*(E)$. Thus,
    \[ m^*(X) = m^*\left(\bigcup_{q \in \mathbb{Q} \cap [-1,1]} q+E\right) = \sum_{q \in \mathbb{Q} \cap [-1,1]} m^*(E). \]
    If $m^*(E) = 0$, then $\sum m^*(E) = 0$. If $m^*(E) > 0$, then $\sum m^*(E) = \infty$. 
    Both cases contradict $1 \le m^*(X) \le 3$. Hence, countable additivity fails for outer measure $m^*$.
\end{pfbox}



\begin{proposition}[Failure of Finite Additivity]
    There exists a finite collection of disjoint subsets of $\mathbb{R}$ such that the outer measure of their union is not equal to the sum of their outer measures.
\end{proposition}

\begin{pfbox}
    We use the same Vitali set $E$ constructed previously. We know $m^*(E) > 0$ because if $m^*(E) = 0$, countable subadditivity would force 
    $$ m^*(X) \le \sum_{q \in \mathbb{Q} \cap [-1,1]} m^*(q+E) = 0, $$ 
    which contradicts $m^*(X) \ge 1$.
    
    Since $m^*(E) > 0$, by the Archimedean property, there exists an integer $n \ge 1$ such that $m^*(E) > \frac{1}{n}$.
    Let $I$ be a finite subset of $\mathbb{Q} \cap [-1,1]$ consisting of exactly $3n$ distinct rational numbers.
    Consider the finite disjoint union $\bigcup_{q \in I} (q+E)$.
    Because this union is a subset of $X$, and we already established $X \subseteq [-1,2]$, we must have:
    $$ m^*\left(\bigcup_{q \in I} (q+E)\right) \le m^*([-1,2]) = 3. $$
    
    Assume for the sake of contradiction that finite additivity holds for outer measure. Then the measure of the union equals the sum of the measures:
    $$ m^*\left(\bigcup_{q \in I} (q+E)\right) = \sum_{q \in I} m^*(q+E) = \sum_{q \in I} m^*(E) = 3n \cdot m^*(E). $$
    Substituting our bound $m^*(E) > \frac{1}{n}$ into the equation, we get:
    $$ 3n \cdot m^*(E) > 3n \left(\frac{1}{n}\right) = 3. $$
    This implies $3 < 3$, which is a blatant contradiction. 
    Thus, finite additivity fails for the Lebesgue outer measure $m^*$.
\end{pfbox}


\subsection{Lebesgue Measure}

\begin{definition}[Lebesgue Measurability]
    Let $E$ be a subset of $\mathbb{R}^n$. We say that $E$ is (Lebesgue) measurable if we have the identity:
    \[ m^*(A) = m^*(A \cap E) + m^*(A \setminus E) \quad \text{for all } A \subseteq \mathbb{R}^n. \]
    In this case, we define the Lebesgue measure of $E$ as $m^*(E)=m(E).$
\end{definition}

\begin{remark}\ \vspace{-1em}
    \begin{enumerate}
        \item This definition is proposed by Constantin Carath\'eodory.
        \item Recall $A = (A \cap E) \cup (A \setminus E)$. By subadditivity, we always have $m^*(A) \le m^*(A \cap E) + m^*(A \setminus E)$. 
            So, $E$ is measurable if and only if $m^*(A) \ge m^*(A \cap E) + m^*(A \setminus E)$.
    \end{enumerate}
\end{remark}

\begin{example}
    $\mathbb{R}^n$ and $\varnothing$ are measurable (Easy exercise).
\end{example}

\begin{proposition}
    The half-space $H = \{\mathbf{x} = (x_1, x_2, \dots, x_n) \in \mathbb{R}^n : x_n \ge 0\}$ is Lebesgue measurable.
\end{proposition}

\begin{pfbox}
    Let $E = H$. For any test set $A \subseteq \mathbb{R}^n$, it suffices to show that $m^*(A) \ge m^*(A \cap E) + m^*(A \setminus E)$.
    
    If $m^*(A) = \infty$, the inequality holds trivially, so there is nothing to prove.
    Next, we assume $m^*(A) < \infty$. Given any $\epsilon > 0$, by the definition of outer measure, there exists a countable collection of open boxes $\{B_k\}_{k=1}^\infty$ such that $A \subseteq \bigcup_{k=1}^\infty B_k$ and
    $$ \sum_{k=1}^\infty \mathrm{Vol}(B_k) \le m^*(A) + \epsilon. $$
    
    For each open box $B_k$, the hyperplane $x_n = 0$ splits it into two pieces: $B_k \cap E$ and $B_k \setminus E$. Since these two pieces are essentially boxes (or empty) and their volumes naturally add up to the volume of the original box $B_k$, we have:
    $$ m^*(B_k) = m^*(B_k \cap E) + m^*(B_k \setminus E). $$
    (Recall that the outer measure of a box is exactly its volume).
    
    Summing this over all $k$, we get:
    $$ \sum_{k=1}^\infty \mathrm{Vol}(B_k) = \sum_{k=1}^\infty m^*(B_k \cap E) + \sum_{k=1}^\infty m^*(B_k \setminus E). $$
    
    By the countable subadditivity of the outer measure, we know:
    $$ \sum_{k=1}^\infty m^*(B_k \cap E) \ge m^*\left( \bigcup_{k=1}^\infty (B_k \cap E) \right) = m^*\left( \left(\bigcup_{k=1}^\infty B_k\right) \cap E \right). $$
    Since $A \subseteq \bigcup B_k$, by monotonicity we have $\left(\bigcup B_k\right) \cap E \supseteq A \cap E$. Therefore:
    $$ m^*\left( \left(\bigcup_{k=1}^\infty B_k\right) \cap E \right) \ge m^*(A \cap E). $$
    Similarly, for the outside part:
    $$ \sum_{k=1}^\infty m^*(B_k \setminus E) \ge m^*(A \setminus E). $$
    
    Substituting these lower bounds back into our volume sum, we obtain:
    $$ m^*(A) + \epsilon \ge \sum_{k=1}^\infty \mathrm{Vol}(B_k) \ge m^*(A \cap E) + m^*(A \setminus E). $$
    Since $\epsilon > 0$ is arbitrary, we let $\epsilon \to 0$ to conclude:
    $$ m^*(A) \ge m^*(A \cap E) + m^*(A \setminus E). $$
    Hence, the half-space $H$ is measurable.
\end{pfbox}

\begin{remark}
    A similar argument can apply to any half-space of the form $\{\mathbf{x} \in \mathbb{R}^n : x_j \ge a\}$ or $\{\mathbf{x} \in \mathbb{R}^n : x_j \le a\}$ for any coordinate $1 \le j \le n$ and any constant $a \in \mathbb{R}$.
\end{remark}

\begin{theorem}[Properties of Measurable Sets]\ \vspace{-1em}
    \begin{enumerate}
        \item[(a)] If $E$ is measurable, then $\mathbb{R}^n \setminus E$ is also measurable.
        \item[(b)] If $E$ is measurable and $x \in \mathbb{R}^n$, then $x+E$ is also measurable.
        \item[(c)] If $E_1$ and $E_2$ are measurable, then $E_1 \cap E_2$ and $E_1 \cup E_2$ are measurable.
        \item[(d)] If $E_1, E_2, \dots, E_N$ are measurable, then $\bigcup_{i=1}^N E_i$ and $\bigcap_{i=1}^N E_i$ are measurable.
        \item[(e)] Every open box and closed box is measurable.
        \item[(f)] Any set $E$ of outer measure zero is measurable.
    \end{enumerate}
\end{theorem}

\begin{pfbox}
    \textbf{Proof of (a):} Since $A \setminus (\mathbb{R}^n \setminus E) = A \cap E$ and $A \cap (\mathbb{R}^n \setminus E) = A \setminus E$, the definition of measurability is symmetric with respect to $E$ and $E^c$.

    \textbf{Proof of (b):} Since $E$ is measurable, to show $x+E$ is measurable, we need to show that for any $A \subseteq \mathbb{R}^n$,
    \[ m^*(A) = m^*(A \cap (x+E)) + m^*(A \setminus (x+E)). \]
    First, observe the following set identities when shifted by $-x$:
    \begin{align*}
        (A \cap (x+E)) - x &= (A - x) \cap E \\
        (A \setminus (x+E)) - x &= (A - x) \setminus E
    \end{align*}
    By the translation invariance of the outer measure $m^*$, we have:
    \begin{align*}
        m^*(A \cap (x+E)) &= m^*((A \cap (x+E)) - x) = m^*((A - x) \cap E) \\
        m^*(A \setminus (x+E)) &= m^*((A \setminus (x+E)) - x) = m^*((A - x) \setminus E)
    \end{align*}
    Let $B = A - x$. Since $E$ is measurable, testing with the set $B$ yields:
    \[ m^*(B) = m^*(B \cap E) + m^*(B \setminus E). \]
    Substituting back $B = A - x$ and using the translation invariance $m^*(A-x) = m^*(A)$ on the left side, we get:
    \[ m^*(A) = m^*((A - x) \cap E) + m^*((A - x) \setminus E). \]
    Replacing the terms on the right side with the equalities we found earlier:
    \[ m^*(A) = m^*(A \cap (x+E)) + m^*(A \setminus (x+E)). \]
    This holds for all $A \subseteq \mathbb{R}^n$, thus $x+E$ is measurable.

\textbf{Proof of (c):} 

    \textit{Part 1: Union is measurable.} \\
    We want to show that $E_1 \cup E_2$ is measurable. For any test set $A \subseteq \mathbb{R}^n$, since $E_1$ is measurable, we can split $A$ using $E_1$:
    $$m^*(A) = m^*(A \cap E_1) + m^*(A \setminus E_1).$$
    Next, we apply the measurability of $E_2$ to the test set $A \setminus E_1$:
    $$m^*(A \setminus E_1) = m^*((A \setminus E_1) \cap E_2) + m^*((A \setminus E_1) \setminus E_2).$$
    Substitute this back into the first equation:
    $$m^*(A) = m^*(A \cap E_1) + m^*((A \setminus E_1) \cap E_2) + m^*((A \setminus E_1) \setminus E_2).$$
    Now, we make two observations from set algebra:
    \begin{itemize}
        \item First, the remaining part outside both sets is $(A \setminus E_1) \setminus E_2 = A \cap E_1^c \cap E_2^c = A \cap (E_1 \cup E_2)^c = A \setminus (E_1 \cup E_2)$.
        \item Second, the parts intersecting the sets combine to $(A \cap E_1) \cup ((A \setminus E_1) \cap E_2) = A \cap (E_1 \cup E_2)$.
    \end{itemize}
    By the subadditivity of the outer measure $m^*$, the sum of the measures of the first two terms is bounded below by the measure of their union:
    $$m^*(A \cap E_1) + m^*((A \setminus E_1) \cap E_2) \ge m^*((A \cap E_1) \cup ((A \setminus E_1) \cap E_2)) = m^*(A \cap (E_1 \cup E_2)).$$
    Replacing these terms in our expanded equation, we obtain:
    $$m^*(A) \ge m^*(A \cap (E_1 \cup E_2)) + m^*(A \setminus (E_1 \cup E_2)).$$
    Since the reverse inequality is always true by subadditivity, $E_1 \cup E_2$ is measurable.

    \textit{Part 2: Intersection is measurable.}

    \textbf{(Method 1: By De Morgan's Laws)} \\
    By De Morgan's laws, we can express the intersection as $E_1 \cap E_2 = (E_1^c \cup E_2^c)^c$.
    Since $E_1$ and $E_2$ are measurable, their complements $E_1^c$ and $E_2^c$ are measurable by Property (a).
    By Part 1 of this proof, the union of two measurable sets is measurable, so $E_1^c \cup E_2^c$ is measurable.
    Finally, taking the complement once more, we conclude that $(E_1^c \cup E_2^c)^c = E_1 \cap E_2$ is measurable.

    \textbf{(Method 2: Direct Carath\'eodory Approach)} \\
    Alternatively, we can prove it directly. Since $E_1$ is measurable, split $A$:
    $$m^*(A) = m^*(A \cap E_1) + m^*(A \setminus E_1).$$
    Apply the measurability of $E_2$ to the test set $A \cap E_1$:
    $$m^*(A \cap E_1) = m^*(A \cap E_1 \cap E_2) + m^*((A \cap E_1) \setminus E_2).$$
    Substitute this into the first equation:
    $$m^*(A) = m^*(A \cap E_1 \cap E_2) + m^*((A \cap E_1) \setminus E_2) + m^*(A \setminus E_1).$$
    Observe that the last two terms cover the part of $A$ outside the intersection: $((A \cap E_1) \setminus E_2) \cup (A \setminus E_1) = A \setminus (E_1 \cap E_2)$.
    By subadditivity, $m^*((A \cap E_1) \setminus E_2) + m^*(A \setminus E_1) \ge m^*(A \setminus (E_1 \cap E_2))$.
    This yields:
    $$m^*(A) \ge m^*(A \cap (E_1 \cap E_2)) + m^*(A \setminus (E_1 \cap E_2)),$$
    which proves $E_1 \cap E_2$ is measurable.

    \textbf{Proof of (d):} Follows from (c) by mathematical induction.

    \textbf{Proof of (e):} \\
    Let $B = \{\mathbf{x} \in \mathbb{R}^n : a_i < x_i < b_i \text{ for } i=1,2,\dots,n\}$ be an open box. 
    We can express $B$ as the intersection of $2n$ half-spaces:
    $$ B = \bigcap_{i=1}^n P_i $$
    where each $P_i = \{\mathbf{x} \in \mathbb{R}^n : x_i > a_i\} \cap \{\mathbf{x} \in \mathbb{R}^n : x_i < b_i\}$.
    From the previous proposition and remark, every such half-space is Lebesgue measurable.
    By Property (d) proved above, any finite intersection of measurable sets is also measurable. 
    Since the open box $B$ is a finite intersection of measurable half-spaces, $B$ is measurable. 
    For any closed box, we can apply the same argument in proof of (c) via De Morgan's Laws, and this is left as an exercise.

    \textbf{Proof of (f):} Suppose $m^*(E) = 0$. For any $A$, $A \cap E \subseteq E \implies m^*(A \cap E) = 0$.
    Also $A \setminus E \subseteq A \implies m^*(A \setminus E) \le m^*(A)$.
    Thus, $m^*(A \cap E) + m^*(A \setminus E) \le 0 + m^*(A) = m^*(A)$. $E$ is measurable.
\end{pfbox}

\begin{theorem}[Finite Additivity]
    If $(E_j)_{j=1}^N$ is a finite collection of disjoint measurable sets, then for any $A \subseteq \mathbb{R}^n$:
    \[ m^*\left(A \cap \bigcup_{j=1}^N E_j\right) = \sum_{j=1}^N m^*(A \cap E_j). \]
    Moreover, taking $A = \mathbb{R}^n$, we have $m\left(\bigcup_{j=1}^N E_j\right) = \sum_{j=1}^N m(E_j)$.
\end{theorem}

\begin{pfbox}
    We proceed by induction. For $N=2$, let $E, F$ be disjoint measurable sets. Test the set $A \cap (E \cup F)$ with the measurable set $E$:
    \begin{align*}
        m^*(A \cap (E \cup F)) &= m^*((A \cap (E \cup F)) \cap E) + m^*((A \cap (E \cup F)) \setminus E) \\
        &= m^*(A \cap E) + m^*(A \cap F).
    \end{align*}
    Suppose it holds for $k$ sets. Let $U_k = \bigcup_{j=1}^k E_j$. Since $U_k \cap E_{k+1} = \varnothing$ and $U_k$ is measurable:
    \[ m^*(A \cap (U_k \cup E_{k+1})) = m^*(A \cap U_k) + m^*(A \cap E_{k+1}) = \sum_{j=1}^k m^*(A \cap E_j) + m^*(A \cap E_{k+1}). \]
\end{pfbox}

\begin{corollary}
    If $A \subseteq B$ are two measurable sets, then $m(B) = m(A) + m(B \setminus A)$.
\end{corollary}

\begin{pfbox}
    Observe that $B\setminus A = B\cap(\mathbb{R}^n\setminus A),$ so $B\setminus A$ is measurable.
    Also, $A$ and $B\setminus A$ are disjoint measurable sets, by finite additivity, we obtain the identity.
\end{pfbox}

\begin{theorem}[Countable Additivity]
    If $(E_j)_{j=1}^\infty$ is a countable collection of disjoint measurable sets, then $E = \bigcup_{j=1}^\infty E_j$ is measurable and $m(E) = \sum_{j=1}^\infty m(E_j)$.
\end{theorem}

\begin{pfbox}
    Let $F_N = \bigcup_{k=1}^N E_k$. $F_N$ is measurable. For any $A \subseteq \mathbb{R}^n$,
    \[ m^*(A) = m^*(A \cap F_N) + m^*(A \setminus F_N) \ge \sum_{k=1}^N m^*(A \cap E_k) + m^*(A \setminus E). \]
    Letting $N \to \infty$, we obtain $m^*(A) \ge \sum_{k=1}^\infty m^*(A \cap E_k) + m^*(A \setminus E)$.
    By subadditivity, $\sum_{k=1}^\infty m^*(A \cap E_k) \ge m^*(A \cap E)$. 
    Thus $m^*(A) \ge m^*(A \cap E) + m^*(A \setminus E)$, proving $E$ is measurable.
    By taking $A = \mathbb{R}^n$, we get $m(F_N) = \sum_{k=1}^N m(E_k) \le m(E)$. Letting $N \to \infty$ gives $\sum_{k=1}^\infty m(E_k) \le m(E)$. The reverse inequality holds by countable subadditivity.
\end{pfbox}

\begin{theorem}[$\sigma$-algebra]
    If $(\Omega_j)_{j=1}^\infty$ is any countable collection of measurable sets, then $\bigcup_{j=1}^\infty \Omega_j$ and $\bigcap_{j=1}^\infty \Omega_j$ are also measurable.
\end{theorem}

\begin{pfbox}
    We can disjointify the union. Let $E_1 = \Omega_1$, and for $N \ge 2$, define
    \[ E_N = \Omega_N \setminus \left(\bigcup_{k=1}^{N-1} \Omega_k\right). \]
    Since finite unions and complements of measurable sets are measurable, each $E_N$ is measurable. 
    The sets $(E_N)$ are pairwise disjoint, and $\bigcup_{j=1}^\infty \Omega_j = \bigcup_{N=1}^\infty E_N$. 
    By the previous theorem, the countable union of disjoint measurable sets is measurable. 
    The intersection is measurable by De Morgan's laws: $\bigcap \Omega_j = \left(\bigcup \Omega_j^c\right)^c$.
\end{pfbox}

\begin{remark}
Every $\sigma$-algebra is a Boolean algebra (finite unions and complements), but the converse is not true.
\end{remark}

\begin{theorem}
    Every open set can be written as a countable union of rational open boxes.
\end{theorem}

\begin{pfbox}
    We say an open box is rational if $B = \prod_{i=1}^n (a_i, b_i)$ where $a_i, b_i \in \mathbb{Q}$. The set of all rational boxes is countable.
    First, we show that for any open ball $B(x,r)$, there exists a rational box $B$ such that $x \in B \subseteq B(x,r)$. 
    Let $a_i, b_i$ be rational numbers such that $x_i - \frac{r}{n} < a_i < x_i < b_i < x_i + \frac{r}{n}$.
    Then $x \in B = \prod (a_i, b_i)$. For any $y \in B$, $|y_i - x_i| < \frac{r}{n}$, which implies $\|y - x\| \le \sum |y_i - x_i| < r$, so $B \subseteq B(x,r)$.
    
    Let $E$ be an open set in $\mathbb{R}^n$. For every $x \in E$, there exists $r_x > 0$ such that $B(x, r_x) \subseteq E$. 
    By the above claim, there is a rational box $B_x$ such that $x \in B_x \subseteq B(x, r_x) \subseteq E$.
    Thus $E = \bigcup_{x \in E} B_x$. Since there are only countably many rational boxes, this union is countable.
\end{pfbox}

\begin{theorem}[Borel property]
    Any open set and closed set is Lebesgue measurable.
\end{theorem}

\begin{pfbox}
    Any open set is the countable union of rational open boxes. Since boxes are measurable and countable unions of measurable sets are measurable, every open set is measurable.
    A closed set is the complement of an open set, hence it is also measurable.
\end{pfbox}


\subsection{Measurable Functions}


\begin{definition}[$\sigma$-Algebra]
    Let $X$ be a set. A collection $\mathcal{A}$ of subsets of $X$ is called a $\sigma$-algebra if it satisfies the following three properties:
    \begin{enumerate}
        \item $X \in \mathcal{A}$.
        \item Closed under complement: If $E \in \mathcal{A}$, then $X \setminus E \in \mathcal{A}$.
        \item Closed under countable unions: If $E_1, E_2, \dots \in \mathcal{A}$, then $\displaystyle\bigcup_{i=1}^\infty E_i \in \mathcal{A}$.
    \end{enumerate}
\end{definition}

\begin{remark}
    The definition of a $\sigma$-algebra inherently implies two additional properties:
    \begin{enumerate}
        \item Contains the empty set: Since $X \in \mathcal{A}$ and it is closed under complement, $X \setminus X = \varnothing \in \mathcal{A}$.
        \item Closed under countable intersections: If $E_1, E_2, \dots \in \mathcal{A}$, their complements $X \setminus E_i$ are also in $\mathcal{A}$. By the countable union property, $\displaystyle\bigcup_{i=1}^\infty (X \setminus E_i) \in \mathcal{A}$. Taking the complement of this union and applying De Morgan's laws yields:
        \[ X \setminus \left( \bigcup_{i=1}^\infty (X \setminus E_i) \right) = \displaystyle\bigcap_{i=1}^\infty E_i \in \mathcal{A}. \]
    \end{enumerate}
\end{remark}

\begin{definition}[Borel $\sigma$-Algebra]
    Let $(X, \mathcal{T})$ be a topological space. The Borel $\sigma$-algebra $\mathcal{B}(X)$ is the $\sigma$-algebra generated by the collection of all open sets $\mathcal{T}$, i.e., $\mathcal{B}(X) = \sigma(\mathcal{T})$.
    
    A subset $E \subseteq X$ is a \textbf{Borel set} if $E \in \mathcal{B}(X)$. 
\end{definition}

\begin{remark}
    Borel $\sigma$-algebra is the smallest $\sigma$-algebra contains all open sets.
    The term ``smallest'' refers to the intersection of all $\sigma$-algebras on $X$ that contain $\mathcal{T}$:
    \[ \sigma(\mathcal{T}) = \bigcap \{ \mathcal{A} : \mathcal{A} \text{ is a } \sigma\text{-algebra on } X \text{ and } \mathcal{T} \subseteq \mathcal{A} \}. \]
    This ensures that $\mathcal{B}(X)$ contains no more sets than are strictly necessary to satisfy the $\sigma$-algebra axioms while including all open sets.
\end{remark}

\begin{example}
    Let $X = \{1, 2, 3\}$ and $\mathcal{T} = \{\{1\}\}$. We want to find the smallest $\sigma$-algebra on $X$ containing $\mathcal{T}$, denoted by $\sigma(\mathcal{T})$.

    According to the axioms of a $\sigma$-algebra, we can construct it step-by-step:
    \begin{enumerate}
        \item \textbf{Include the empty set and the whole space:} 
        A $\sigma$-algebra must contain $\varnothing$ and $X$. \\
        Current collection: $\{ \varnothing, \{1, 2, 3\} \}$
        
        \item \textbf{Include the generating sets:} 
        The $\sigma$-algebra must contain all elements of $\mathcal{T}$. \\
        Current collection: $\{ \varnothing, \{1, 2, 3\}, \{1\} \}$
        
        \item \textbf{Close under complements:} 
        The complement of $\{1\}$ is $X \setminus \{1\} = \{2, 3\}$. This must also be included. \\
        Current collection: $\{ \varnothing, \{1, 2, 3\}, \{1\}, \{2, 3\} \}$
        
        \item \textbf{Close under countable unions:} 
        We check all possible unions of the sets in our collection. For instance, $\{1\} \cup \{2, 3\} = \{1, 2, 3\}$, which is already in our collection. No new sets can be generated.
    \end{enumerate}

    Therefore, the smallest $\sigma$-algebra generated by $\mathcal{T}$ is:
    \[ \sigma(\mathcal{T}) = \{ \varnothing, \{1, 2, 3\}, \{1\}, \{2, 3\} \}. \]
\end{example}

\begin{definition}[Measurable Function]
    Let $\Omega$ be a measurable subset of $\mathbb{R}^n$ and let $f:\Omega\rightarrow \mathbb{R}^m$ be a function. We say $f$ is measurable if $f^{-1}(V)$ is measurable for every open set $V \subseteq \mathbb{R}^m$.
\end{definition}

\begin{remark}
    In particular, we call $f$ is Borel measurable since $f^{-1}(B)$ is measurable for every Borel set $B \in \mathcal{B}(\mathbb{R}^m).$
\end{remark}



\begin{theorem}
    Let $f: \Omega \subseteq \mathbb{R}^n \rightarrow \mathbb{R}^m$ be a continuous function, where $\Omega$ is measurable. Then $f$ is measurable.
\end{theorem}

\begin{pfbox}
    Let $W$ be an open set in $\mathbb{R}^m$. Since $f$ is continuous, $f^{-1}(W) = \Omega \cap U$, where $U$ is an open set in $\mathbb{R}^n$. Because $U$ is open, it is measurable. Since $\Omega$ is measurable, their intersection $\Omega \cap U$ is also measurable. Thus, $f^{-1}(W)$ is measurable, which means $f$ is measurable.
\end{pfbox}

\begin{theorem}
    Let $\Omega$ be a measurable subset in $\mathbb{R}^n$ and $f: \Omega \rightarrow \mathbb{R}^m$ be a function. $f$ is measurable if and only if $f^{-1}(B)$ is measurable for every open box $B$.
\end{theorem}

\begin{pfbox}
    $(\Rightarrow)$ An open box $B$ is an open set, so $f^{-1}(B)$ is measurable by definition.
    
    $(\Leftarrow)$ Given any open set $V$, it can be written as a countable union of open boxes: $V = \bigcup_{i=1}^\infty B_i$. Recall that $f^{-1}\left(\bigcup_{i=1}^\infty B_i\right) = \bigcup_{i=1}^\infty f^{-1}(B_i)$. Since each $f^{-1}(B_i)$ is measurable by assumption, and countable union of measurable sets is measurable, $f^{-1}(V)$ is measurable.
\end{pfbox}

\begin{corollary}
    Let $\Omega$ be a measurable subset of $\mathbb{R}^n$, and let $f:\Omega\rightarrow \mathbb{R}^m$ be a function such that $f = (f_1, f_2, \dots, f_m)$, where $f_j:\Omega\rightarrow \mathbb{R}$ is the $j$-th coordinate of $f$. Then $f$ is measurable if and only if all of the $f_j$ are individually measurable.
\label{coordinatewise measurable}
\end{corollary}

\begin{pfbox}
    $(\Rightarrow)$ Fix $j \in \{1, 2, \dots, m\}$ and consider $f_j : \Omega \rightarrow \mathbb{R}$. Let $I \subseteq \mathbb{R}$ be an open interval. Consider the open set $B_j = \mathbb{R} \times \mathbb{R} \times \dots \times I \times \dots \times \mathbb{R}$ (where $I$ is at the $j$-th position). Note that $x \in f^{-1}(B_j) \Leftrightarrow f(x) \in B_j \Leftrightarrow f_j(x) \in I \Leftrightarrow x \in f_j^{-1}(I)$. If $f$ is measurable, $f^{-1}(B_j)$ is measurable, so $f_j^{-1}(I)$ is measurable for any open interval $I$, hence $f_j$ is measurable.
    
    $(\Leftarrow)$ Let $B = I_1 \times I_2 \times \dots \times I_m$ be an open box in $\mathbb{R}^m$. $x \in f^{-1}(B) \Leftrightarrow f(x) \in B \Leftrightarrow f_j(x) \in I_j$ for $j=1, \dots, m \Leftrightarrow x \in f_j^{-1}(I_j)$ for $j=1, \dots, m$. Thus $f^{-1}(B) = \bigcap_{j=1}^m f_j^{-1}(I_j)$. Since each $f_j$ is measurable, $f_j^{-1}(I_j)$ is measurable, so their finite intersection $f^{-1}(B)$ is measurable.
\end{pfbox}

\begin{theorem}
    Let $\Omega$ be a measurable subset of $\mathbb{R}^n$, and let $W$ be an open subset of $\mathbb{R}^m$. If $f: \Omega \rightarrow W$ is measurable, and $g: W \rightarrow \mathbb{R}^p$ is continuous, then $g \circ f: \Omega \rightarrow \mathbb{R}^p$ is measurable.
\label{composition measurable}
\end{theorem}

\begin{pfbox}
    Let $V$ be an open set in $\mathbb{R}^p$. We have $(g \circ f)^{-1}(V) = f^{-1}(g^{-1}(V))$. Since $g$ is continuous, $g^{-1}(V) = W \cap U$, where $U$ is open by the continuity of $g$. So $g^{-1}(V)$ is open in $W$. Because $f$ is measurable, $f^{-1}(g^{-1}(V))$ is measurable. Thus $(g \circ f)^{-1}(V)$ is measurable.
\end{pfbox}

\begin{corollary}
    Let $\Omega\subseteq\mathbb{R}^n$ be measurable. If $f:\Omega\to\mathbb{R}$ is measurable,
    then $|f|, \max\{f,0\}, \min\{f,0\}$ is measurable.
\end{corollary}

\begin{pfbox}
    Let \vspace{-1em}
    \begin{enumerate}
        \item $g_1: \mathbb{R}\to\mathbb{R}$ defined by $g_1(x)=|x|,$
        \item $g_2: \mathbb{R}\to\mathbb{R}$ defined by $g_2(x)=\frac{x+|x|}{2},$
        \item $g_3: \mathbb{R}\to\mathbb{R}$ defined by $g_3(x)=\frac{x-|x|}{2}.$
    \end{enumerate}
    where $g_1, g_2, g_3$ are continous on their open domains.
    By theorem \eqref{composition measurable}, we are done.
\end{pfbox}

\begin{corollary}
    If $f, g: \Omega \rightarrow \mathbb{R}$ are measurable functions, then $f+g$, $f-g$, $fg$, $f/g$ (where $g \neq 0$), $\max\{f,g\}$, and $\min\{f,g\}$ are measurable.
\end{corollary}

\begin{pfbox}
    Consider $F: \Omega \rightarrow \mathbb{R}^2$ defined by $F(x) = (f(x), g(x))$. By Corollary \eqref{coordinatewise measurable}, $f, g$ are measurable $\Rightarrow F$ is measurable. 
    Define 
    \begin{enumerate}
        \item $h_{\pm}: \mathbb{R}^2\to\mathbb{R}$ defined by $h_{\pm}(u,v) = u \pm v,$
        \item $h_{*}: \mathbb{R}^2\to\mathbb{R}$ defined by $h_{*}(u,v)=uv,$
        \item $h_{\div}: \mathbb{R}^2\setminus\{0\}\to\mathbb{R}$ defined by $h_{\div}(u,v) = u/v,$
        \item $h_{\max}: \mathbb{R}^2\to\mathbb{R}$ defined by $h_{\max}(u,v) = \frac{u+v+|u-v|}{2},$
        \item $h_{\min}: \mathbb{R}^2\to\mathbb{R}$ defined by $h_{\min}(u,v) = \frac{u+v-|u-v|}{2}.$
    \end{enumerate}
    The above functions are all continuous on therie respective open domains. By theorem \eqref{composition measurable}, we are done.
\end{pfbox}

\begin{proposition}
    Let $\Omega$ be a measurable subset of $\mathbb{R}^n$, $f:\Omega \rightarrow \mathbb{R}$ be a function. $f$ is measurable if and only if $f^{-1}((\alpha, \infty))$ is measurable for any $\alpha \in \mathbb{R}$.
\end{proposition}

\begin{pfbox}
    $(\Rightarrow)$ Trivial, because $(\alpha, \infty)$ is an open set.
    
    $(\Leftarrow)$ Let $I = (\alpha, \beta)$ be an open interval. We can write $(\alpha, \beta) = (\alpha, \infty) \cap (\mathbb{R} \setminus [\beta, \infty))$. Note that $[\beta, \infty) = \bigcap_{m=1}^\infty (\beta - \frac{1}{m}, \infty)$. 
    Therefore, $f^{-1}([\beta, \infty)) = f^{-1}\left(\bigcap_{m=1}^\infty (\beta - \frac{1}{m}, \infty)\right) = \bigcap_{m=1}^\infty f^{-1}((\beta - \frac{1}{m}, \infty))$. By assumption, $f^{-1}((\beta - \frac{1}{m}, \infty))$ is measurable, so the countable intersection is measurable. 
    Thus, $f^{-1}((\alpha, \beta)) = f^{-1}((\alpha, \infty)) \cap f^{-1}(\mathbb{R} \setminus [\beta, \infty))$ is the intersection of two measurable sets, hence measurable.
\end{pfbox}

\begin{definition}[Measurable Functions in the Extended Reals]
    Let $\Omega$ be a measurable subset of $\mathbb{R}^n$. The set of \textbf{extended real numbers} is defined as:
    \[ \mathbb{R}^* = \mathbb{R} \cup \{ \infty, -\infty \}. \]
    A function $f: \Omega \to \mathbb{R}^*$ is called \textbf{measurable} if the set $f^{-1}((a, \infty])$ is measurable for all $a \in \mathbb{R}$.
\end{definition}

\begin{remark}
    Note that $f^{-1}((a, \infty]) = f^{-1}((a, \infty)) \cup f^{-1}(\{\infty\})$. This definition implies that $f^{-1}((a, b))$ is measurable for any $a < b$, because we can write the interval as:
    \[ (a, b) = (a, \infty] \cap [-\infty, b) = (a, \infty] \cap \left( \mathbb{R}^* \setminus [b, \infty] \right). \]
    The set $[b, \infty]$ is measurable because it can be expressed as a countable intersection:
    \[ [b, \infty] = \displaystyle\bigcap_{n=1}^\infty \left( b - \frac{1}{n}, \infty \right]. \]
\end{remark}


\begin{theorem}
    Let $\Omega \subseteq \mathbb{R}^n$ be measurable. For each positive integer $n$, let $f_n: \Omega \rightarrow \mathbb{R}^*$ be a measurable function. Then $\sup f_n$, $\inf f_n$, $\limsup_{n\to\infty} f_n$, and $\liminf_{n\to\infty} f_n$ are measurable.
\end{theorem}

\begin{pfbox}
    Let $g(x) = \sup_{n \ge 1} f_n(x)$. We check if $g^{-1}((a, \infty])$ is measurable for any $a \in \mathbb{R}$. 
    $x \in g^{-1}((a, \infty]) \Leftrightarrow \sup f_n(x) > a \Leftrightarrow \exists\, k \in \mathbb{N} \text{ s.t. } f_k(x) > a \Leftrightarrow x \in f_k^{-1}((a, \infty])$. 
    Therefore, $g^{-1}((a, \infty]) = \displaystyle\bigcup_{n=1}^\infty f_n^{-1}((a, \infty])$, which is a countable union of measurable sets, so it is measurable.
    
    Let $h(x) = \inf_{n \ge 1} f_n(x)$. $x \in h^{-1}([a, \infty]) \Leftrightarrow \inf f_n(x) \ge a \Leftrightarrow \forall n \in \mathbb{N}, f_n(x) \ge a \Leftrightarrow x \in \displaystyle\bigcap_{n=1}^\infty f_n^{-1}([a, \infty])$. This is a countable intersection of measurable sets, so it is measurable.
    
    Since $\displaystyle\limsup_{n\to\infty} f_n(x) = \displaystyle\inf_{N \ge 1} \sup_{n \ge N} f_n(x)$, let $s_N(x)=\displaystyle\sup_{n\ge N}f_n(x)$, we know $s_N$ is measurable for each $N$. After taking the infimum, $\displaystyle\limsup_{n\to\infty} f_n(x)$ is measurable. Similarly for $\liminf$.
\end{pfbox}

\begin{remark}
    In particular, if a sequence of measurable functions $\{f_n\}$ converges pointwise to a function $f$ (i.e., $\lim_{n \to \infty} f_n(x) = f(x)$ for all $x \in \Omega$), then $f$ is also measurable. This follows immediately from the theorem, because if the limit exists, we have:
    \[ f(x) = \limsup_{n \to \infty} f_n(x) = \liminf_{n \to \infty} f_n(x). \]
    This is one of the most powerful properties of Lebesgue measurable functions: the pointwise limit of a sequence of measurable functions is always measurable (unlike continuous functions, where a pointwise limit may fail to be continuous).
\end{remark}


\subsection{Lebesgue Integration}

\begin{definition}[Characteristic Function]
    Let $\Omega$ be a set and $E \subseteq \Omega$ be a subset. The \textbf{characteristic function} of $E$, denoted by $\chi_E: \Omega \rightarrow \mathbb{R}$, is defined as:
    \[ 
    \chi_E(x) = \begin{cases} 
    1 & \text{if } x \in E \\ 
    0 & \text{if } x \in \Omega \setminus E 
    \end{cases}
    \]
\end{definition}

\begin{remark}
    In some contexts, the characteristic function is also called the \textbf{indicator function}, often denoted by $\mathds{1}_E(x)$. Note that $\chi_E$ maps the membership of a set to the numerical values $1$ and $0$, which allows us to convert set operations into algebraic operations. For example:
    \begin{enumerate}
        \item $\chi_{A \cap B}(x) = \chi_A(x) \cdot \chi_B(x)$
        \item $\chi_{A \cup B}(x) = \chi_A(x) + \chi_B(x) - \chi_{A \cap B}(x)$
        \item $\chi_{\Omega \setminus E}(x) = 1 - \chi_E(x)$
    \end{enumerate}
\end{remark}

\begin{proposition}
    Let $\Omega \subseteq \mathbb{R}^n$ be a measurable set and let $E$ be a measurable subset of $\Omega$. The characteristic function $\chi_E : \Omega \rightarrow \mathbb{R}$ is measurable.
\end{proposition}

\begin{pfbox}
    Let $I$ be an open interval in $\mathbb{R}$. 
    If $0 \notin I, 1 \notin I$, then $\chi_E^{-1}(I) = \varnothing$.
    If $0 \in I, 1 \notin I$, then $\chi_E^{-1}(I) = \Omega \setminus E$.
    If $0 \notin I, 1 \in I$, then $\chi_E^{-1}(I) = E$.
    If $0 \in I, 1 \in I$, then $\chi_E^{-1}(I) = \Omega$.
    In all cases, the resulting set is measurable. Thus $\chi_E$ is measurable.
\end{pfbox}




\begin{definition}[Simple Function]
    Let $\Omega \subseteq \mathbb{R}^n$ be a measurable set, and $f:\Omega\rightarrow \mathbb{R}$ be a measurable function. We say $f$ is a simple function if the image $f(\Omega)$ is finite.
\end{definition}



\begin{proposition}
    Let $f: \Omega \rightarrow \mathbb{R}$ be a simple function (a measurable function where $f(\Omega)$ is finite). Then there exist finitely many real numbers $c_1, \dots, c_N$ and pairwise disjoint measurable sets $E_1, \dots, E_N$ such that $f = \displaystyle\sum_{j=1}^N c_j \chi_{E_j}$.
\end{proposition}

\begin{pfbox}
    By definition of simple function, $f(\Omega)$ is finite. 
    WLOG, we may write the image as $f(\Omega) = \{c_1, c_2, \dots, c_N\}$ where $c_j$ are distinct. Define $E_j = f^{-1}(\{c_j\})$ for each $1 \le j \le N$. 
    Suppose $E_m \cap E_n \neq \varnothing$ for $m \neq n$. Then $\exists p \in E_m \cap E_n$ such that $f(p) = c_m$ and $f(p) = c_n$, which violates the definition of a function since $c_m \neq c_n$. So $E_j$ are disjoint.
    Also, $\{c_j\} = \displaystyle\bigcap_{k=1}^\infty \left(c_j - \frac{1}{k}, c_j + \frac{1}{k}\right)$, so $E_j = f^{-1}(\{c_j\}) = \displaystyle\bigcap_{k=1}^\infty f^{-1}\left(\left(c_j - \frac{1}{k}, c_j + \frac{1}{k}\right)\right)$, which is a countable intersection of measurable sets and thus measurable.
    Note that $\Omega = f(\cup\{c_j\})=\cup f(\{c_j\})=\cup E_j.$ For any $x \in \Omega$, $x \in E_k$ for some $k$, so $f(x) = c_k$. Simultaneously, $\displaystyle\sum_{j=1}^N c_j \chi_{E_j}(x) = c_k \chi_{E_k}(x) = c_k$.
\end{pfbox}

\begin{theorem}
    Let $\Omega$ be a measurable subset of $\mathbb{R}^n$, and let $f:\Omega\rightarrow[0, \infty]$ be a non-negative measurable function. Then there exists a sequence of simple functions $f_1, f_2, \dots$ such that each $f_n$ is non-negative, the sequence is increasing ($0 \le f_1 \le f_2 \le \dots$), and $\displaystyle\lim_{n\to\infty} f_n(x) = f(x)$ for all $x \in \Omega$.
\end{theorem}

\begin{pfbox}
    For each integer $n$, define $E_{n,k} = \left\{x \in \Omega : \frac{k}{2^n} \le f(x) < \frac{k+1}{2^n}\right\}$ for $k = 0, 1, \dots, n2^n-1$, and $F_n = \{x \in \Omega : f(x) \ge n\}$. Both $E_{n,k}$ and $F_n$ are pre-images of intervals, hence measurable.
    Define $f_n(x) = \displaystyle\sum_{k=0}^{n2^n-1} \frac{k}{2^n} \chi_{E_{n,k}}(x) + n \chi_{F_n}(x)$. 
    To show $f_n$ is increasing, look at $f_{n+1}(x)$:
    If $f(x) \ge n$, then $f_n(x) = n \le f_{n+1}(x)$.
    If $f(x) < n$, then for some $k$, $\frac{k}{2^n} \le f(x) < \frac{k+1}{2^n}$, so $f_n(x) = \frac{k}{2^n}$. In the $n+1$ grid, this interval is split into $\left[\frac{2k}{2^{n+1}}, \frac{2k+1}{2^{n+1}}\right)$ and $\left[\frac{2k+1}{2^{n+1}}, \frac{2(k+1)}{2^{n+1}}\right)$. So $f_{n+1}(x)$ is either $\frac{2k}{2^{n+1}} = \frac{k}{2^n}$ or $\frac{2k+1}{2^{n+1}} > \frac{k}{2^n}$. Thus $f_n(x) \le f_{n+1}(x)$.
    As $n \to \infty$, if $f(x) = \infty$, $f_n(x) = n \to \infty$.\\
    If $f(x) < \infty\,\exists\,N\in\mathbb{N}$ s.t. $f(x)<N$. Let $n \ge f(x)$, there exists some $k$ s.t. 
    \begin{align*}
        \frac{k}{2^n}\le f(x) < \frac{k+1}{2^n} &\Rightarrow f_n(x)\le f(x)<f_n(x)+\frac{1}{2^n}\\
        &\Rightarrow 0 \le |f(x) - f_n(x)| < \frac{1}{2^n},
    \end{align*}
    so $\displaystyle\lim_{n\to\infty} f_n(x) = f(x)$.
\end{pfbox}

\begin{definition}[Lebesgue Integral of Non-Negative Simple Function]
    Let $\Omega \subseteq \mathbb{R}^n$ be measurable, and $f:\Omega \rightarrow \mathbb{R}$ be a simple function, where $f \ge 0$. Thus, $f$ is measurable and the image $f(\Omega)$ is finite and contained in $[0, \infty)$. We define the Lebesgue integral of $f$ on $\Omega$ by
    \[ \displaystyle\int_{\Omega} f = \displaystyle\sum_{\lambda \in f(\Omega), \lambda > 0} \lambda \, m(\{x \in \Omega : f(x) = \lambda\}). \]
\end{definition}

\begin{remark}
    By definition, a non-negative simple function is a finite linear combination of characteristic functions:
    \[ f = \displaystyle\sum_{j=1}^N c_j \chi_{E_j}, \]
    where the coefficients $c_j$ are distinct non-negative real numbers and the corresponding measurable sets $E_j = \{x \in \Omega : f(x) = c_j\}$ are pairwise disjoint. 
    Using this explicit form, the Lebesgue integral of $f$ evaluates directly to:
    \[ \displaystyle\int_{\Omega} f = \displaystyle\sum_{j=1}^N c_j m(E_j) \]
    Note that if any coefficient $c_j = 0$, its contribution to the sum is $0 \cdot m(E_j) = 0$. 
\end{remark}

\begin{proposition}
    Let $\Omega \subseteq \mathbb{R}^n$ be measurable, and let $E_1, \dots, E_N$ be pairwise disjoint measurable subsets of $\Omega$. Let $c_1, \dots, c_N$ be non-negative real numbers (may repeat). Then 
    $$\int_\Omega \sum_{j=1}^N c_j \chi_{E_j} = \sum_{j=1}^N c_j m(E_j).$$
\end{proposition}

\begin{pfbox}
    Let $f = \displaystyle\sum_{j=1}^N c_j \chi_{E_j}$. Let $d_1, d_2, \dots, d_L$ be the distinct positive numbers among $c_1, \dots, c_N$. Define $F_l = \displaystyle\bigcup_{\{j: c_j = d_l\}} E_j$. Since $E_j$ are disjoint and measurable, $m(F_l) = \displaystyle\sum_{\{j: c_j = d_l\}} m(E_j)$. 
    We can rewrite $f$ as $f=\displaystyle\sum_{l=1}^L d_l \chi_{F_l}$, where $d_l$ are distinct.
    By the definition of the integral of a simple function, 
    $$\int_\Omega f = \sum_{l=1}^L d_l m(F_l) = \sum_{l=1}^L d_l \sum_{\{j:\, c_j = d_l\}} m(E_j) = \sum_{l=1}^L\sum_{\{j:\, c_j = d_l\}} d_l m(E_j) = \sum_{j=1}^N c_j m(E_j).$$
\end{pfbox}


\begin{theorem}
    Let $\Omega \subseteq \mathbb{R}^n$ be a measurable set, and let $f$ and $g$ be non-negative simple functions. The following properties hold:
    \begin{itemize}
        \item[(a)] $0 \le \int_\Omega f \le \infty$. Furthermore, $\int_\Omega f = 0$ if and only if $m(\{x \in \Omega : f(x) \neq 0\}) = 0$.
        \item[(b)] $\int_\Omega (f+g) = \int_\Omega f + \int_\Omega g$.
        \item[(c)] $\int_\Omega cf = c \int_\Omega f$ for any constant $c \ge 0$.
        \item[(d)] If $f(x) \le g(x)$ for all $x \in \Omega$, then $\int_\Omega f \le \int_\Omega g$.
        \item[(e)] If $f(x) = g(x)$ almost everywhere, then $\int_\Omega f = \int_\Omega g$.
    \end{itemize}
\end{theorem}

\begin{pfbox}
    \textbf{Proof of (a):} If $f=0$ on $\Omega$, then clearly $\int_\Omega f = 0$. In general, we may assume $f = \displaystyle\sum_{j=1}^N c_j \chi_{E_j}$ with $c_j \ge 0$. By definition, $\int_\Omega f = \displaystyle\sum_{j=1}^N c_j m(E_j)$, which implies $0 \le \int_\Omega f \le \infty$ (the integral reaches infinity if $m(E_j) = \infty$ for some $j$ where $c_j > 0$).
    
    $(\Rightarrow)$ Suppose $\int_\Omega f = 0$. Since all terms $c_j m(E_j)$ are non-negative, we must have $c_j m(E_j) = 0$ for all $j=1,\dots,N$. For any $c_k > 0$, it follows that $m(E_k) = 0$. Therefore, $m(\{x \in \Omega : f(x) \neq 0\}) = m\left(\displaystyle\bigcup_{\{k: c_k > 0\}} E_k\right) = \displaystyle\sum_{\{k: c_k > 0\}} m(E_k) = 0$.
    
    $(\Leftarrow)$ Conversely, suppose $m(\{x \in \Omega : f(x) \neq 0\}) = 0$. For any $k$ such that $c_k > 0$, we have $E_k \subseteq \{x \in \Omega : f(x) \neq 0\}$, which implies $m(E_k) = 0$. Thus, every term in the sum $\displaystyle\sum_{j=1}^N c_j m(E_j)$ is zero, yielding $\int_\Omega f = 0$.

    \textbf{Proof of (b):} Let $f = \displaystyle\sum_{j=0}^N c_j \chi_{E_j}$ and $g = \displaystyle\sum_{k=0}^M d_k \chi_{F_k}$, where $\{c_0, \dots, c_N\}$ and $\{d_0, \dots, d_M\}$ are distinct non-negative real numbers, and $\{E_j\}_{j=0}^N$ and $\{F_k\}_{k=0}^M$ are disjoint measurable sets forming partitions of $\Omega$.
    The intersection $\{E_j \cap F_k\}$ forms a finite, pairwise disjoint measurable partition of $\Omega$. We can express the indicator functions as $\chi_{E_j} = \displaystyle\sum_{k=0}^M \chi_{E_j \cap F_k}$ and $\chi_{F_k} = \displaystyle\sum_{j=0}^N \chi_{E_j \cap F_k}$.
    This allows us to write $f + g = \displaystyle\sum_{j=0}^N \displaystyle\sum_{k=0}^M (c_j + d_k) \chi_{E_j \cap F_k}$.
    Taking the integral yields:
    \begin{align*}
        \int_\Omega (f+g) &= \displaystyle\sum_{j=0}^N \displaystyle\sum_{k=0}^M (c_j + d_k) m(E_j \cap F_k) \\
        &= \displaystyle\sum_{j=0}^N c_j m\left(\displaystyle\bigcup_{k=0}^M (E_j \cap F_k)\right) + \displaystyle\sum_{k=0}^M d_k m\left(\displaystyle\bigcup_{j=0}^N (E_j \cap F_k)\right) \\
        &= \displaystyle\sum_{j=0}^N c_j m(E_j) + \displaystyle\sum_{k=0}^M d_k m(F_k) = \int_\Omega f + \int_\Omega g.
    \end{align*}

    \textbf{Proof of (c):} By direct computation, 
    $$\int_\Omega cf = \displaystyle\sum_{j=1}^N (c \cdot c_j) m(E_j) = c \displaystyle\sum_{j=1}^N c_j m(E_j) = c \int_\Omega f.$$

    \textbf{Proof of (d):} Since $f(x) \le g(x)$ for all $x \in \Omega$, the difference $g - f$ is a non-negative simple function. We can write $g = (g - f) + f$. Applying property (b), we obtain $\int_\Omega g = \int_\Omega (g - f) + \int_\Omega f$. Since the integral of a non-negative simple function is non-negative, $\int_\Omega g \ge \int_\Omega f$.

    \textbf{Proof of (e):} Let $N = \{x \in \Omega : f(x) \neq g(x)\} = \{x \in \Omega : (f - g)(x) \neq 0\}$. By the given assumption, $m(N) = 0$, $f=g$ almost everywhere.
    Since $f$ and $g$ are simple functions, $f-g$ is also a simple function. Applying the result from (a), since the measure of the set where $f-g \neq 0$ is zero (i.e. $f-g=0$ a.e.), we have $\int_\Omega f - g = 0$. Consequently, $\int_\Omega f = \int_\Omega g$ by (b).
\end{pfbox}

\begin{definition}[Lebesgue Integral for Non-Negative Measurable Functions]
    Let $\Omega \subseteq \mathbb{R}^n$ be a measurable set and let $f: \Omega \rightarrow [0, \infty]$ be a measurable and non-negative function. We define the Lebesgue integral of $f$ over $\Omega$ as the supremum of the integrals of all non-negative simple functions bounded by $f$:
    $$ \int_\Omega f = \sup \left\{ \int_\Omega s : 0 \le s \le f, \text{ where } s \text{ is a simple function} \right\} $$
\end{definition}

\begin{remark}\ \vspace{-1em}
    \begin{enumerate}
        \item If $\Omega' \subseteq \Omega$ is a measurable subset, then the integral over $\Omega'$ is defined by $\int_{\Omega'} f := \int_{\Omega'} f|_{\Omega'}$.
        \item If $f$ itself is a non-negative simple function such that $f = \displaystyle\sum_{j=1}^N c_j \chi_{E_j}$, this supremum definition coincides with the standard simple function integral $\int_\Omega f = \displaystyle\sum_{j=1}^N c_j m(E_j)$.
        \item By definition, the value of the integral satisfies $0 \le \int_\Omega f \le \infty$.
    \end{enumerate}
\end{remark}

\begin{theorem}
    Let $\Omega \subseteq \mathbb{R}^n$ be measurable, and let $f, g: \Omega \rightarrow [0, \infty]$ be non-negative measurable functions. The following properties hold:
    \begin{itemize}
        \item[(a)] $0 \le \int_\Omega f \le \infty$. Furthermore, $\int_\Omega f = 0$ iff $f(x) = 0$ a.e.
        \item[(b)] $\int_\Omega cf = c \int_\Omega f$ for any constant $c \ge 0$.
        \item[(c)] If $f(x) \le g(x)$ for all $x \in \Omega$, then $\int_\Omega f \le \int_\Omega g$.
        \item[(d)] If $f(x) = g(x)$ almost everywhere in $\Omega$, then $\int_\Omega f = \int_\Omega g$.
        \item[(e)] If $\Omega'$ is a measurable subset of $\Omega$, then $\int_{\Omega'} f = \int_\Omega f \chi_{\Omega'} \le \int_\Omega f$.
    \end{itemize}
\end{theorem}

\begin{pfbox}
    \textbf{Proof of (a):} 
    By definition, $\int_\Omega f = \sup \{ \int_\Omega s : 0 \le s \le f \}$. Since $s$ is non-negative, $0 \le \int_\Omega s$, which directly implies $0 \le \sup \int_\Omega s \le \infty$.
    
    $(\Rightarrow)$ Assume $\int_\Omega f = 0$. We wish to show that $f=0$ almost everywhere, which is equivalent to proving that the measure of the set $E = \{x \in \Omega : f(x) > 0\}$ is zero.
    We can write $E$ as a countable union $E = \displaystyle\bigcup_{k=1}^\infty A_k$, where $A_k = \{x \in \Omega : f(x) \ge \frac{1}{k}\}$. Because $f$ is measurable, each $A_k = f^{-1}([\frac{1}{k}, \infty])$ is a measurable set.
    Define a simple function $s_k = \frac{1}{k} \chi_{A_k}$. It is clear that $s_k \le f$. Thus, $\int_\Omega s_k \le \int_\Omega f = 0$. Since $\int_\Omega s_k = \frac{1}{k} m(A_k)$, it must be that $\frac{1}{k} m(A_k) = 0$, implying $m(A_k) = 0$ for all positive integers $k$.
    By countable subadditivity, $m(E) \le \displaystyle\sum_{k=1}^\infty m(A_k) = 0$. Therefore, $f(x) = 0$ almost everywhere.
    
    $(\Leftarrow)$ Assume $f(x) = 0$ almost everywhere. Let $E = \{x \in \Omega : f(x) > 0\}$, so $m(E) = 0$. Let $s$ be any non-negative simple function such that $s \le f$. For any $x \in \Omega \setminus E$, we have $f(x) = 0$, and therefore $s(x) = 0$.
    We can express $s = \displaystyle\sum_{i=1}^N c_i \chi_{E_i}$. If $c_i > 0$, then $E_i$ must be a subset of $E$, which means $m(E_i) \le m(E) = 0$. Thus, $\int_\Omega s = \displaystyle\sum c_i m(E_i) = 0$. Since the integral of every minorizing simple function is zero, the supremum $\int_\Omega f$ must also be zero.

    \textbf{Proof of (b):} Let $c > 0$. We first show $\int_\Omega cf \ge c \int_\Omega f$. Let $s$ be any simple function with $0 \le s \le f$. Multiplying by $c$ yields $0 \le cs \le cf$. Since $cs$ is a simple function bounded by $cf$, we have $\int_\Omega cs \le \int_\Omega cf$. Therefore, $c \int_\Omega s \le \int_\Omega cf$. Taking the supremum over all such $s$ yields $c \int_\Omega f \le \int_\Omega cf$.
    Next, we show $\int_\Omega cf \le c \int_\Omega f$. Let $t$ be any simple function such that $0 \le t \le cf$. Dividing by $c$, we get $0 \le \frac{t}{c} \le f$. Thus, $\int_\Omega \frac{t}{c} \le \int_\Omega f$, which implies $\int_\Omega t \le c \int_\Omega f$. Taking the supremum over all such $t$ yields $\int_\Omega cf \le c \int_\Omega f$. This establishes equality.

    \textbf{Proof of (c):} Consider any simple function $s$ such that $0 \le s \le f$. Since $f \le g$, it follows that $0 \le s \le g$. Therefore, $\int_\Omega s \le \int_\Omega g$. Because this holds for all simple functions bounded by $f$, taking the supremum yields $\int_\Omega f \le \int_\Omega g$.

    \textbf{Proof of (d):} Let $N = \{x \in \Omega : f(x) \neq g(x)\}$. By assumption, $m(N) = 0$. Consequently, $f = g$ on the set $\Omega \setminus N$.
    We first prove $\int_\Omega f \le \int_\Omega g$. Let $s$ be any simple function bounded by $f$ ($0 \le s \le f$). Define a new simple function $t = s \chi_{\Omega \setminus N}$.
    For any $x \in \Omega \setminus N$, $t(x) = s(x) \le f(x) = g(x)$. For any $x \in N$, $t(x) = 0 \le g(x)$.
    Thus, $0 \le t \le g$ everywhere on $\Omega$, which implies $\int_\Omega t \le \int_\Omega g$. Since $t = s$ almost everywhere (they differ only on $N$), we know from the simple function properties that $\int_\Omega t = \int_\Omega s$. Hence, $\int_\Omega s \le \int_\Omega g$. Taking the supremum over $s$ yields $\int_\Omega f \le \int_\Omega g$.
    A completely symmetric argument proves $\int_\Omega g \le \int_\Omega f$, thus confirming equality.

    \textbf{Proof of (e):} We want to show $\int_{\Omega'} f = \int_\Omega f \chi_{\Omega'}$. Let $s$ be any simple function on $\Omega'$ such that $0 \le s \le f|_{\Omega'}$. We can extend $s$ to a function $\tilde{s}$ on all of $\Omega$ by setting $\tilde{s}(x) = s(x)$ on $\Omega'$ and $0$ on $\Omega \setminus \Omega'$.
    It is clear that $\tilde{s} \le f \chi_{\Omega'}$, which implies $\int_\Omega \tilde{s} \le \int_\Omega f \chi_{\Omega'}$. Since $\int_{\Omega} \tilde{s} = \int_{\Omega'} s$, we have $\int_{\Omega'} s \le \int_\Omega f \chi_{\Omega'}$. Taking the supremum over $s$ gives $\int_{\Omega'} f \le \int_\Omega f \chi_{\Omega'}$.
    Conversely, let $t$ be any simple function on $\Omega$ such that $0 \le t \le f \chi_{\Omega'}$. We notice that $t$ restricted to $\Omega'$ satisfies $t = t|_\Omega'$ (Why?), $t|_{\Omega'} \le f \chi_{\Omega'} = f|_{\Omega'}$. Therefore, $\int_\Omega t = \int_{\Omega'} t|_{\Omega'} \le \int_{\Omega'} f|_{\Omega'} = \int_{\Omega'} f$. Taking the supremum over $t$ yields $\int_\Omega f \chi_{\Omega'} \le \int_{\Omega'} f$.
    Finally, since $f \chi_{\Omega'} \le f$, property (c) guarantees that $\int_\Omega f \chi_{\Omega'} \le \int_\Omega f$, giving us the final bound $\int_{\Omega'} f \le \int_\Omega f$.
\end{pfbox}

    It is very difficult to prove that $\int_\Omega (f+g) = \int_\Omega f + \int_\Omega g$ directly. 
    We require a big theroem to prove this later.

\begin{definition}[Absolutely Integrable Function]
    Let $\Omega \subseteq \mathbb{R}^n$ be measurable and $f : \Omega \rightarrow \mathbb{R}^*$ be a measurable function. We say $f$ is absolutely integrable if the integral of its absolute value is finite, i.e., $\int_\Omega |f| < \infty$.
\end{definition}

\begin{definition}
    Let $\Omega \subseteq \mathbb{R}^n$ be a measurable set. We define $\mathcal{L}^p(\Omega)$ as the space of all measurable functions $f: \Omega \rightarrow \mathbb{R}^*$ such that $\int_\Omega |f|^p d\mu < \infty$. 
\end{definition}
    However, $\mathcal{L}^p(\Omega)$ is only a seminormed space, not a true normed space, because $\int_\Omega |f|^p d\mu = 0$ only implies $f = 0$ almost everywhere, not everywhere. To resolve this, we introduce an equivalence relation $f \sim g$ if $f = g$ a.e. 

\begin{definition}
    The Lebesgue space $L^p(\Omega)$ is defined as the quotient space $\mathcal{L}^p(\Omega) / \sim$, which consists of equivalence classes of functions:
    $$ L^p(\Omega) = \left\{ [f] \mid f \in \mathcal{L}^p(\Omega) \right\} $$
    equipped with the norm $\|f\|_{L^p} = \left( \int_\Omega |f|^p d\mu \right)^{1/p}$.     
\end{definition}

\begin{remark}\ \vspace{-1em}
    \begin{enumerate}
        \item In particular, for $p=1$, the $L^1$ space is the set of all equivalence classes of absolutely integrable functions, with the norm $\|f\|_{L^1} = \int_\Omega |f| d\mu$.
        \item Strictly speaking, the elements of $L^1(\Omega)$ are equivalence classes $[f]$, not individual functions $f$. However, it is a universal and standard abuse of terminology in mathematical analysis to simply write $f \in L^1(\Omega)$ instead of $[f] \in L^1(\Omega)$, and to treat the elements of $L^p$ spaces as functions. We do this with the implicit understanding that we consider two functions to be "the exact same" if they are equal almost everywhere.
    \end{enumerate}
\end{remark}

If $f: \Omega \rightarrow \mathbb{R}$ is a generic measurable function, it might take negative values. We can decompose it into a positive part and a negative part:
$f^+(x) = \max\{f(x), 0\}$, and $f^-(x) = \max\{-f(x), 0\}$.
Both $f^+$ and $f^-$ are non-negative. Obviously, we can write $f(x) = f^+(x) - f^-(x)$ and the absolute value as $|f(x)| = f^+(x) + f^-(x)$.
Since $f$ is measurable, $f^+$, $f^-$, and $|f|$ are all measurable. By the additivity of non-negative integrals, $\int_\Omega |f| = \int_\Omega f^+ + \int_\Omega f^-$.
If a function is absolutely integrable ($\int_\Omega |f| < \infty$), both $\int_\Omega f^+ < \infty$ and $\int_\Omega f^- < \infty$ must hold. This allows us to define the Lebesgue integral of $f$ unambiguously.

\begin{definition}[Lebesgue Integral of an Absolutely Integrable Function]
    Let $f \in L^1(\Omega)$. We define the Lebesgue integral of $f$ as $\int_\Omega f = \int_\Omega f^+ - \int_\Omega f^-$. This prevents the indeterminate form $\infty - \infty$.
\end{definition}

\begin{theorem}
    Let $\Omega \subseteq \mathbb{R}^n$ be a measurable set, $c \in \mathbb{R}$ a constant, and let $f,g \in L^1(\Omega)$. The following properties hold:
    \begin{itemize}
        \item[(a)] $cf \in L^1(\Omega)$ and $\int_\Omega cf = c \int_\Omega f$ 
        \item[(b)] $f+g \in L^1(\Omega)$ and $\int_\Omega (f+g) = \int_\Omega f + \int_\Omega g$
        \item[(c)] $f-g \in L^1(\Omega)$ and $\int_\Omega (f-g) = \int_\Omega f - \int_\Omega g$
        \item[(d)] If $f(x) \le g(x)$ for all $x \in \Omega$, then $\int_\Omega f \le \int_\Omega g$
        \item[(e)] $|\int_\Omega f| \le \int_\Omega |f|$
        \item[(f)] If $f(x) = g(x)$ almost everywhere, then $\int_\Omega f = \int_\Omega g$
    \end{itemize}
\end{theorem}

\begin{pfbox}
    \textbf{Proof of (a):} We check absolute integrability: $\int_\Omega |cf| = \int_\Omega |c||f| = |c| \int_\Omega |f| < \infty$. Hence, $cf \in L^1(\Omega)$.
    If $c \ge 0$, then the positive and negative parts scale cleanly: $(cf)^+ = c f^+$ and $(cf)^- = c f^-$.
    Thus, $\int_\Omega cf = \int_\Omega (cf)^+ - \int_\Omega (cf)^- = c \int_\Omega f^+ - c \int_\Omega f^- = c \int_\Omega f$.
    If $c < 0$, the scaling reverses the parts: $(cf)^+ = (-c) f^-$ and $(cf)^- = (-c) f^+$.
    Thus, $\int_\Omega cf = \int_\Omega (-c) f^- - \int_\Omega (-c) f^+ = (-c) \int_\Omega f^- - (-c) \int_\Omega f^+ = c \int_\Omega f^+ - c \int_\Omega f^- = c \int_\Omega f$.

    \textbf{Proof of (b):} By the triangle inequality, $|f+g| \le |f| + |g|$. Integrating gives $\int_\Omega |f+g| \le \int_\Omega (|f|+|g|) = \int_\Omega |f| + \int_\Omega |g| < \infty$, guaranteeing $f+g \in L^1(\Omega)$.
    Observe the algebraic identity $f+g = (f^+ - f^-) + (g^+ - g^-) = (f^+ + g^+) - (f^- + g^-)$. All terms in parentheses are non-negative.
    By definition, $\int_\Omega (f+g) = \int_\Omega (f^+ + g^+) - \int_\Omega (f^- + g^-)$. Utilizing the additivity for non-negative functions, we get $\int_\Omega f^+ + \int_\Omega g^+ - \int_\Omega f^- - \int_\Omega g^-$, which regroups exactly to $\int_\Omega f + \int_\Omega g$.

    \textbf{Proof of (c):} This is a direct consequence of (a) and (b), since $f-g = f + (-1)g$.

    \textbf{Proof of (d):} Since $g-f \in L^1(\Omega)$ and $g-f \ge 0$, its integral must be non-negative: $\int_\Omega (g-f) \ge 0$. By linearity, $\int_\Omega g - \int_\Omega f \ge 0$, which means $\int_\Omega f \le \int_\Omega g$.

    \textbf{Proof of (e):} Since $f \in L^1(\Omega)$, both $|f|$ and $-|f|$ are in $L^1(\Omega)$. Because $-|f| \le f \le |f|$, applying property (d) preserves the inequalities under the integral sign: $-\int_\Omega |f| \le \int_\Omega f \le \int_\Omega |f|$. This is the definition of absolute value bounds, giving $|\int_\Omega f| \le \int_\Omega |f|$.

    \textbf{Proof of (f):} Define $h = f-g \in L^1(\Omega)$. Since $f=g$ almost everywhere, $h=0$ a.e. and $|h|=0$ a.e. Therefore, $\int_\Omega |h| = 0$ implies $\int_\Omega h = 0$. By linearity, $\int_\Omega f - \int_\Omega g = 0$.

\end{pfbox}



\subsection{LMCT, Fatuo and LDCT}



\begin{theorem}[Lebesgue Monotone Convergence Theorem]
    Let $\Omega \subseteq \mathbb{R}^n$ be a measurable set, and let $\{f_n\}$ be a sequence of non-negative measurable functions, $f_n: \Omega \rightarrow [0, \infty]$. Suppose the sequence is monotonically increasing, meaning $f_n(x) \le f_{n+1}(x)$ for all $x \in \Omega$ and all positive integers $n$.
    Then, the limit function $f(x) = \displaystyle\lim_{n\to\infty} f_n(x)$ is measurable, and we can interchange the limit and the integral:
    $$ \int_\Omega \lim_{n\to\infty} f_n = \lim_{n\to\infty} \int_\Omega f_n $$
\end{theorem}

\begin{pfbox}
    Let $f(x) = \sup_n f_n(x) = \displaystyle\lim_{n\to\infty} f_n(x)$ (the limit exists because the sequence is monotonically increasing). Note that $f: \Omega \rightarrow [0, \infty]$ is non-negative and measurable. We want to show that $\int_\Omega f = \displaystyle\sup_n \int_\Omega f_n$.

    \textbf{Step 1: Bounding the sequence of integrals.} 
    Since $f_n(x) \le f_{n+1}(x)$, property (c) tells us that $\int_\Omega f_n \le \int_\Omega f_{n+1}$. Thus, the sequence of real numbers $\{\int_\Omega f_n\}$ is monotonically increasing. Furthermore, for all $n$, $f_n \le f$, which implies $\int_\Omega f_n \le \int_\Omega f$. Taking the supremum over $n$, we obtain $\displaystyle\sup_n \int_\Omega f_n \le \int_\Omega f$.

    \textbf{Step 2: Proving the reverse inequality.} 
    Recall that $\int_\Omega f = \sup_{0 \le s \le f} \int_\Omega s$, where $s$ is a simple function. It suffices to show that for any simple function $s$ with $0 \le s \le f$, we have $\int_\Omega s \le \sup_n \int_\Omega f_n$.
    Let $0 < \epsilon < 1$ be given. For each positive integer $n$, define the set $E_n := \{x \in \Omega : f_n(x) \ge (1-\epsilon)s(x)\}$. Since $f_n$ and $s$ are measurable functions, $E_n$ is a measurable set. Because the sequence $\{f_n\}$ is monotonically increasing, it follows that $E_n \subseteq E_{n+1}$.
    
    We claim that the union $\displaystyle\bigcup_{n=1}^\infty E_n = \Omega$.
    To prove this claim, fix any $x \in \Omega$. By definition, $0 \le s(x) \le f(x) = \sup_n f_n(x)$. We need to show there exists an index $N$ such that $f_N(x) \ge (1-\epsilon)s(x)$.
    Suppose by contradiction that $f_n(x) < (1-\epsilon)s(x)$ for all $n \in \mathbb{N}$. If $s(x) = 0$, this yields $0 \le f_n(x) < 0$, which is impossible. If $s(x) > 0$, taking the supremum yields $f(x) = \sup_n f_n(x) \le (1-\epsilon)s(x) < s(x)$. This contradicts the fact that $s(x) \le f(x)$. Hence, the claim holds.

    \textbf{Step 3: Convergence of the simple function integral.} 
    Our goal is to show that $\displaystyle\lim_{n\to\infty} \int_{E_n} s = \int_\Omega s$.
    If $s = 0$, the equality is trivial. Assume $s \neq 0$, and write it in its standard representation $s = \displaystyle\sum_{j=1}^N c_j \chi_{F_j}$.
    The integral of $s$ over $E_n$ is $\int_{E_n} s = \int_\Omega s \chi_{E_n} = \int_\Omega \displaystyle\sum_{j=1}^N c_j \chi_{F_j \cap E_n} = \displaystyle\sum_{j=1}^N c_j m(F_j \cap E_n)$.
    Notice that the sets $F_j \cap E_n$ are nested ($F_j \cap E_1 \subseteq F_j \cap E_2 \dots$), and their union over $n$ is $F_j \cap \Omega = F_j$. By the continuity of measure from below, $\displaystyle\lim_{n\to\infty} m(F_j \cap E_n) = m\left(\displaystyle\bigcup_{n=1}^\infty (F_j \cap E_n)\right) = m(F_j)$.
    Therefore, $\displaystyle\lim_{n\to\infty} \int_{E_n} s = \displaystyle\sum_{j=1}^N c_j m(F_j) = \int_\Omega s$.

    \textbf{Step 4: Final comparison.} 
    By the definition of $E_n$, we know that on $E_n$, $f_n(x) \ge (1-\epsilon)s(x)$. Integrating over $E_n$ gives $(1-\epsilon) \int_{E_n} s \le \int_{E_n} f_n$. Since $f_n$ is non-negative, $\int_{E_n} f_n \le \int_\Omega f_n$.
    Taking the supremum over all $n$ on both sides:
    $(1-\epsilon) \displaystyle\lim_{n\to\infty} \int_{E_n} s \le \sup_n \int_\Omega f_n$.
    Using the result from Step 3, we have $(1-\epsilon) \int_\Omega s \le \sup_n \int_\Omega f_n$.
    Since this inequality holds for any $0 < \epsilon < 1$, we can let $\epsilon \to 0$ to obtain $\int_\Omega s \le \sup_n \int_\Omega f_n$.
    Finally, since this holds for every simple function $0 \le s \le f$, taking the supremum over all such $s$ yields $\int_\Omega f \le \sup_n \int_\Omega f_n$. This completes the proof.
\end{pfbox}

\begin{remark}
    The Monotone Convergence Theorem implies that we can interchange limit and integration if the sequence is increasing. In general, we cannot arbitrarily interchange limit and integration.
    For example, let $f_n(x) = \chi_{[n, n+1]}$. For all $n$, $\int_\mathbb{R} f_n = 1$, which means $\lim_{n\to\infty} \int_\mathbb{R} f_n = 1$. However, the pointwise limit is $\lim_{n\to\infty} f_n(x) = 0$, so $\int_\mathbb{R} (\lim_{n\to\infty} f_n) = 0$. The reason MCT fails here is that the sequence $f_n$ is not monotonically increasing.
\end{remark}

\begin{theorem}
    Let $\Omega \subseteq \mathbb{R}^n$ be measurable, and let $f,g : \Omega \rightarrow [0, \infty]$ be measurable functions. Then $\int_\Omega (f+g) = \int_\Omega f + \int_\Omega g$.
\end{theorem}
\begin{pfbox}
    Because $f$ and $g$ are non-negative measurable functions, there exist sequences of simple functions $\{s_n\}$ and $\{t_n\}$ such that $s_n \uparrow f$ and $t_n \uparrow g$.
    It follows that the sequence $(s_n + t_n)$ is monotonically increasing and converges pointwise to $(f+g)$.
    Applying the Lebesgue Monotone Convergence Theorem, we have $\displaystyle\lim_{n\to\infty} \int_\Omega (s_n + t_n) = \int_\Omega (f+g)$.
    Since $s_n$ and $t_n$ are simple functions, we already know that $\int_\Omega (s_n + t_n) = \int_\Omega s_n + \int_\Omega t_n$ for all $n$.
    Taking the limit: $\displaystyle\lim_{n\to\infty} \left(\int_\Omega s_n + \int_\Omega t_n\right) = \displaystyle\lim_{n\to\infty} \int_\Omega s_n + \displaystyle\lim_{n\to\infty} \int_\Omega t_n = \int_\Omega f + \int_\Omega g$. Thus, the equality holds.
\end{pfbox}

\begin{corollary}
    If $\Omega \subseteq \mathbb{R}^n$ is measurable, and $g_1, g_2, \dots$ are non-negative measurable functions from $\Omega \rightarrow [0, \infty]$, then the integral of the infinite sum equals the infinite sum of the integrals: $\int_\Omega \displaystyle\sum_{n=1}^\infty g_n = \displaystyle\sum_{n=1}^\infty \int_\Omega g_n$.
\end{corollary}
\begin{pfbox}
    Let the partial sum be $G_N = \displaystyle\sum_{n=1}^N g_n$. The sequence $G_N$ is non-negative, monotonically increasing, and measurable.
    By the Lebesgue Monotone Convergence Theorem, $\displaystyle\lim_{N\to\infty} \int_\Omega G_N = \int_\Omega \lim_{N\to\infty} G_N = \int_\Omega \displaystyle\sum_{n=1}^\infty g_n$.
    On the other hand, evaluating the limit directly using finite additivity:
    $\displaystyle\lim_{N\to\infty} \int_\Omega G_N = \displaystyle\lim_{N\to\infty} \int_\Omega \displaystyle\sum_{n=1}^N g_n = \displaystyle\lim_{N\to\infty} \displaystyle\sum_{n=1}^N \int_\Omega g_n = \displaystyle\sum_{n=1}^\infty \int_\Omega g_n$.
\end{pfbox}

\begin{lemma}[Fatou]
    Let $\Omega \subseteq \mathbb{R}^n$ be a measurable set, and let $f_1, f_2, \dots : \Omega \rightarrow [0, \infty]$ be a sequence of non-negative measurable functions. Then,
    $$ \int_\Omega \liminf_{n\to\infty} f_n \le \liminf_{n\to\infty} \int_\Omega f_n $$
\end{lemma}
\begin{pfbox}
    Define a new sequence of functions $g_N(x) = \inf_{n \ge N} f_n(x)$. As $N$ increases, the set of functions we are taking the infimum over shrinks, which means $g_N$ is monotonically increasing.
    By the Lebesgue Monotone Convergence Theorem, we can interchange the limit and the integral for $g_N$: $\displaystyle\lim_{N\to\infty} \int_\Omega g_N = \int_\Omega \lim_{N\to\infty} g_N$.
    By definition, the limit of $g_N$ is exactly the limit infimum: $\displaystyle\lim_{N\to\infty} g_N = \liminf_{n\to\infty} f_n$.
    Thus, $\int_\Omega \liminf_{n\to\infty} f_n = \displaystyle\lim_{N\to\infty} \int_\Omega g_N$.
    
    Now, fix a specific $N \in \mathbb{N}$. For all $n \ge N$, we have $g_N(x) = \inf_{k \ge N} f_k(x) \le f_n(x)$.
    Integrating both sides gives $\int_\Omega g_N(x) \le \int_\Omega f_n(x)$ for all $n \ge N$. Taking the infimum over $n \ge N$ on the right side yields $\int_\Omega g_N(x) \le \inf_{n \ge N} \int_\Omega f_n(x)$.
    Finally, taking the limit as $N \to \infty$ on both sides:
    $\displaystyle\lim_{N\to\infty} \int_\Omega g_N(x) \le \lim_{N\to\infty} \inf_{n \ge N} \int_\Omega f_n(x) = \liminf_{n\to\infty} \int_\Omega f_n$.
    Combining this with our earlier equality completes the proof.
\end{pfbox}

\begin{lemma}
    Let $\Omega \subseteq \mathbb{R}^n$ be measurable, and let $f : \Omega \rightarrow [0, \infty]$ be a measurable function such that its integral is finite ($\int_\Omega f < \infty$). Then $f$ is finite almost everywhere, meaning the set $\{x \in \Omega : f(x) = \infty\}$ has measure zero.
\end{lemma}
\begin{pfbox}
    For each integer $k \ge 1$, define the set $A_k := \{x \in \Omega : f(x) > k\}$. Because $f$ is measurable, the preimage $A_k = f^{-1}((k, \infty])$ is a measurable set.
    The set of points where $f$ is infinity can be written as an infinite intersection: $\{x \in \Omega : f(x) = \infty\} = \displaystyle\bigcap_{k=1}^\infty A_k$.
    Define the characteristic function scaled by $k$: $k \chi_{A_k}$.
    If $x \in A_k$, then $k \chi_{A_k}(x) = k < f(x)$. If $x \notin A_k$, then $k \chi_{A_k}(x) = 0 \le f(x)$. In either case, the simple function $k \chi_{A_k}$ is bounded by $f$.
    Therefore, $\int_\Omega k \chi_{A_k} \le \int_\Omega f$, which simplifies to $k \cdot m(A_k) \le \int_\Omega f$, or $m(A_k) \le \frac{1}{k} \int_\Omega f$.
    Since the integral $\int_\Omega f$ is a finite constant, taking the limit as $k \to \infty$ gives $\displaystyle\lim_{k\to\infty} m(A_k) = 0$.
    Since the sets are nested ($A_{k+1} \subset A_k$), the measure of their intersection is the limit of their measures: $m\left(\displaystyle\bigcap_{k=1}^\infty A_k\right) = \displaystyle\lim_{k\to\infty} m(A_k) = 0$.
\end{pfbox}

\begin{lemma}[Borel-Cantelli]
    Let $\Omega_1, \Omega_2, \dots$ be measurable subsets of $\mathbb{R}^n$ such that the sum of their measures is finite ($\displaystyle\sum_{n=1}^\infty m(\Omega_n) < \infty$). Then the set $N = \{x \in \mathbb{R}^n : x \in \Omega_n \text{ for infinitely many } n\}$ has measure zero.
\end{lemma}
\begin{pfbox}
    Let $G_N(x) = \displaystyle\sum_{n=1}^N \chi_{\Omega_n}(x)$ be the number of sets among the first $N$ sets that contain $x$. $G_N$ is a non-negative, monotonically increasing, measurable function.
    Integrating $G_N$ yields $\int_\Omega G_N = \displaystyle\sum_{n=1}^N \int_\Omega \chi_{\Omega_n} = \displaystyle\sum_{n=1}^N m(\Omega_n) < \infty$.
    By the Lebesgue Monotone Convergence Theorem, we can pass the limit inside the integral: $\displaystyle\lim_{N\to\infty} \int_\Omega G_N = \int_\Omega \lim_{N\to\infty} G_N = \int_\Omega \displaystyle\sum_{n=1}^\infty \chi_{\Omega_n} = \displaystyle\sum_{n=1}^\infty m(\Omega_n) < \infty$.
    Let $G(x) = \displaystyle\lim_{N\to\infty} G_N(x) = \displaystyle\sum_{n=1}^\infty \chi_{\Omega_n}(x)$. The function $G(x)$ counts the total number of sets $\Omega_n$ that contain $x$.
    If a point $x$ belongs to infinitely many sets $\Omega_n$, then $G(x) = \infty$. Thus, the set $N$ is contained in the set $\{x \in \mathbb{R}^n : G(x) = \infty\}$.
    Because we have established that $\int_\Omega G < \infty$, the previous lemma guarantees that $G(x)$ can only be infinity on a set of measure zero. Therefore, $N$ must have measure zero.
\end{pfbox}



\begin{theorem}[Lebesgue Dominated Convergence Theorem]
    Let $\Omega \subseteq \mathbb{R}^n$ be measurable, and let $\{f_n\}$ be a sequence of measurable functions from $\Omega$ to $\mathbb{R}^*$ such that the pointwise limit $\displaystyle\lim_{n\to\infty} f_n(x) = f(x)$ exists for almost every $x \in \Omega$.
    Furthermore, suppose there exists an absolutely integrable function $F: \Omega \rightarrow [0, \infty]$ that strictly dominates the sequence: $|f_n(x)| \le F(x)$ for almost every $x \in \Omega$. Then we can interchange the limit and the integral:
    $$ \int_\Omega f = \lim_{n\to\infty} \int_\Omega f_n $$
\end{theorem}

\begin{pfbox}
    First, we discard the problematic sets. Let $E = \{x : F(x) = \infty\}$. Since $F$ is absolutely integrable ($\int_\Omega |F| < \infty$), the measure $m(E) = 0$.
    We can restrict our attention to the domain $\Omega \setminus E$, where $F(x) < \infty$. The integrals over $\Omega \setminus E$ are identical to the integrals over $\Omega$. On this restricted domain, $f_n$ takes values in $\mathbb{R}$.
    The domination condition $|f_n(x)| \le F(x)$ is equivalent to $-F(x) \le f_n(x) \le F(x)$. This gives us two non-negative sequences of functions: $F(x) + f_n(x) \ge 0$ and $F(x) - f_n(x) \ge 0$.
    
    We apply Fatou's Lemma to the non-negative sequence $\{F + f_n\}$:
    $$ \int_\Omega \liminf_{n\to\infty} (F+f_n) \le \liminf_{n\to\infty} \int_\Omega (F+f_n) $$
    Because $F$ is independent of $n$ and integrable, we can pull it out:
    $$ \int_\Omega (F+f) \le \int_\Omega F + \liminf_{n\to\infty} \int_\Omega f_n $$
    Subtracting the finite value $\int_\Omega F$ from both sides yields: $\int_\Omega f \le \liminf_{n\to\infty} \int_\Omega f_n$.

    Next, we apply Fatou's Lemma to the non-negative sequence $\{F - f_n\}$:
    $$ \int_\Omega \liminf_{n\to\infty} (F-f_n) \le \liminf_{n\to\infty} \int_\Omega (F-f_n) $$
    Notice that $\liminf (F - f_n) = F - \limsup f_n = F - f$. The right side becomes $\liminf (\int F - \int f_n) = \int F - \limsup \int f_n$.
    Substituting these into the inequality:
    $$ \int_\Omega (F-f) \le \int_\Omega F - \limsup_{n\to\infty} \int_\Omega f_n $$
    Subtracting $\int_\Omega F$ and multiplying by $-1$ reverses the inequality: $\int_\Omega f \ge \limsup_{n\to\infty} \int_\Omega f_n$.
    
    Combining our two results, we have established:
    $$ \limsup_{n\to\infty} \int_\Omega f_n \le \int_\Omega f \le \liminf_{n\to\infty} \int_\Omega f_n $$
    Since the limit infimum can never exceed the limit supremum ($\liminf \int f_n \le \limsup \int f_n$), they must all be strictly equal. Therefore, the limit exists and $\displaystyle\lim_{n\to\infty} \int_\Omega f_n = \int_\Omega f$.
\end{pfbox}


\begin{definition}[Upper and Lower Lebesgue Integral]
    Let $\Omega \subseteq \mathbb{R}^n$ be measurable, and let $f: \Omega \rightarrow \mathbb{R}$ be a function (which is NOT necessarily measurable).
    We define the upper Lebesgue integral as the infimum of the integrals of all bounding $L^1$ functions from above:
    $$ \overline{\int}_\Omega f := \inf \left\{ \int_\Omega g : g \in L^1(\Omega), \text{ and } g \ge f \right\} $$
    Similarly, the lower Lebesgue integral is defined using functions bounding from below:
    $$ \underline{\int}_\Omega f := \sup \left\{ \int_\Omega h : h \in L^1(\Omega), \text{ and } h \le f \right\} $$
\end{definition}

\begin{remark}
    By definition, the lower integral is always less than or equal to the upper integral: $\underline{\int}_\Omega f \le \overline{\int}_\Omega f$. If $f \in L^1(\Omega)$, then we can just pick $f$ itself as the bounding function, meaning $\underline{\int}_\Omega f = \int_\Omega f = \overline{\int}_\Omega f$.
\end{remark}

\begin{theorem}
    Let $\Omega \subseteq \mathbb{R}^n$ be measurable and let $f: \Omega \rightarrow \mathbb{R}$ be a function. If the upper and lower integrals match ($\overline{\int}_\Omega f = \underline{\int}_\Omega f$), then $f \in L^1(\Omega)$ (meaning $f$ is guaranteed to be measurable and absolutely integrable), and $\int_\Omega f = \overline{\int}_\Omega f$.
\end{theorem}
\begin{pfbox}
    Assume the integrals match and denote the common value by $A = \overline{\int}_\Omega f = \underline{\int}_\Omega f$.
    By the infimum definition of the upper integral, for each positive integer $n \ge 1$, there exists an upper bounding function $\tilde{f}_n \in L^1(\Omega)$ such that $\tilde{f}_n \ge f$ and $\int_\Omega \tilde{f}_n \le A + \frac{1}{n}$.
    Similarly, by the supremum definition of the lower integral, there exists a lower bounding function $\underaccent{\tilde}{f}_n \in L^1(\Omega)$ such that $\underaccent{\tilde}{f}_n \le f$ and $A - \frac{1}{n} \le \int_\Omega \underaccent{\tilde}{f}_n$.
    
    We construct the pointwise infimum and supremum functions: define $\tilde{F} = \inf_n \tilde{f}_n$ and $\underaccent{\tilde}{F} = \sup_n \underaccent{\tilde}{f}_n$.
    Because taking the infimum or supremum preserves inequalities, we have the tight sandwich: $\underaccent{\tilde}{F} \le f \le \tilde{F}$. Since countable infimums and supremums of measurable functions are measurable, $\tilde{F}$ and $\underaccent{\tilde}{F}$ are measurable.
    Furthermore, they are absolutely integrable because they are bounded by the first terms of our sequences: $|\tilde{F}| \le |\tilde{f}_1|$ and $|\underaccent{\tilde}{F}| \le |\underaccent{\tilde}{f}_1|$, which guarantees $\int_\Omega |\tilde{F}| \le \int_\Omega |\tilde{f}_1| < \infty$. Thus $\tilde{F}, \underaccent{\tilde}{F} \in L^1(\Omega)$.
    
    Since $\tilde{F} \le \tilde{f}_n$, we know $\int_\Omega \tilde{F} \le A + \frac{1}{n}$ for all $n$. Letting $n \to \infty$ gives $\int_\Omega \tilde{F} \le A$.
    Similarly, since $\underaccent{\tilde}{F} \ge \underaccent{\tilde}{f}_n$, we know $A - \frac{1}{n} \le \int_\Omega \underaccent{\tilde}{F}$ for all $n$, yielding $A \le \int_\Omega \underaccent{\tilde}{F}$.
    Because $\underaccent{\tilde}{F} \le \tilde{F}$, we must also have $\int_\Omega \underaccent{\tilde}{F} \le \int_\Omega \tilde{F}$. The only way to satisfy all these inequalities is if $\int_\Omega \tilde{F} = \int_\Omega \underaccent{\tilde}{F} = A$.
    
    This implies $\int_\Omega (\tilde{F} - \underaccent{\tilde}{F}) = 0$. Since $\tilde{F} - \underaccent{\tilde}{F} \ge 0$, the function must be zero almost everywhere, meaning $\tilde{F} = \underaccent{\tilde}{F}$ almost everywhere.
    Since $f$ is sandwiched between them ($\underaccent{\tilde}{F} \le f \le \tilde{F}$), it must be that $\underaccent{\tilde}{F} = f = \tilde{F}$ almost everywhere.
    Because $f$ is equal to an $L^1$ function almost everywhere, $f$ is measurable, $f \in L^1(\Omega)$, and its integral is exactly $A$.
\end{pfbox}

\begin{theorem}
    Let $I$ be a bounded interval and $f: I \rightarrow \mathbb{R}$ be a Riemann integrable function. Then $f$ is also (absolutely) Lebesgue integrable, and the Lebesgue integral equals the Riemann integral: $\int_I f d\mu = \mathcal{R}\int_I f dx$.
\end{theorem}
\begin{pfbox}
    Let $A = \mathcal{R}\int_I f$. Because $f$ is Riemann integrable, for any $\epsilon > 0$, there exists a partition $\mathcal{P}$ of the interval $I$ into finitely many subintervals $J$ such that the lower and upper Darboux sums are bounded tight around $A$:
    $$ A - \epsilon \le \displaystyle\sum_{J \in \mathcal{P}} |J| \inf_{x \in J} f(x) \le A \le \displaystyle\sum_{J \in \mathcal{P}} |J| \sup_{x \in J} f(x) \le A + \epsilon $$
    We can construct step functions (which are essentially simple functions) based on this partition. Define the lower step function $f_\epsilon = \displaystyle\sum_{J \in \mathcal{P}} \inf_{x \in J} f(x) \chi_J$ and the upper step function $\tilde{f}_\epsilon = \displaystyle\sum_{J \in \mathcal{P}} \sup_{x \in J} f(x) \chi_J$.
    Obviously, $f_\epsilon \le f \le \tilde{f}_\epsilon$. Their Lebesgue integrals perfectly match the Darboux sums, yielding:
    $$ A - \epsilon \le \int_I f_\epsilon \le A \le \int_I \tilde{f}_\epsilon \le A + \epsilon $$
    Because $f_\epsilon$ bounds $f$ from below, its integral bounds the lower Lebesgue integral: $\int_I f_\epsilon \le \underline{\int}_I f$. Because $\tilde{f}_\epsilon$ bounds $f$ from above, its integral bounds the upper Lebesgue integral: $\overline{\int}_I f \le \int_I \tilde{f}_\epsilon$.
    Stringing these inequalities together gives $A - \epsilon \le \underline{\int}_I f \le \overline{\int}_I f \le A + \epsilon$.
    Since this bound holds for any arbitrary $\epsilon > 0$, we conclude that the upper and lower integrals are exactly equal to $A$ ($\underline{\int}_I f = \overline{\int}_I f = A$). By the previous theorem, $f$ is Lebesgue measurable, and its Lebesgue integral equals $A$.
\end{pfbox}

Lastly, we shall prove a big theorem in Calculus.
Suppose $f$ is an absolutely integrable function on $\mathbb{R}^2$. We would like to calculate the double integral by evaluating iterated integrals. Fubini's Theorem guarantees that under absolute integrability, the order of integration does not matter:
$$ \int_{\mathbb{R}^2} f = \int_\mathbb{R} \int_\mathbb{R} f(x,y) dx dy = \int_\mathbb{R} \int_\mathbb{R} f(x,y) dy dx $$

\begin{lemma}[Fubini's Theorem for Characteristic Functions]
    Let $Q = [-N,N]^2$ for some integer $N \in \mathbb{N}$, and let $E \subseteq Q$ be a measurable set. Then for almost every $x \in [-N,N]$, the slice function $\chi_E(x,y)$ is integrable with respect to $y$, and for almost every $y \in [-N,N]$, $\chi_E(x,y)$ is integrable with respect to $x$. Furthermore, we have:
    $$ \int_{[-N,N]} \left( \int_{[-N,N]} \chi_E(x,y) dy \right) dx = \int_{[-N,N]} \left( \int_{[-N,N]} \chi_E(x,y) dx \right) dy = \int_{Q} \chi_E = m(E) $$
\end{lemma}

\begin{pfbox}
    \textbf{Step 1: The key identity for characteristic functions.}
    Because $\chi_E$ is not guaranteed to be measurable over individual slices initially, we define the upper and lower integrals over the $y$-slice.
    For each $x \in [-N,N]$, define $U(x) = \overline{\int}_{[-N,N]} \chi_E(x,y)dy$ and $L(x) = \underline{\int}_{[-N,N]} \chi_E(x,y)dy$.
    We claim that the upper integral of $U(x)$ is bounded by the measure of $E$: $\overline{\int}_{[-N,N]} U(x)dx \le m(E)$. (This will be proven in Step 2).
    
    If this claim is true, the same logic applied to the complement $Q \setminus E$ gives $\overline{\int}_{[-N,N]} \overline{\int}_{[-N,N]} \chi_{Q \setminus E}(x,y)dydx \le m(Q \setminus E) = 4N^2 - m(E)$.
    However, the left-hand side can be rewritten: $\overline{\int} \overline{\int} (1 - \chi_E)dydx = 4N^2 - \underline{\int} \underline{\int} \chi_E dydx = 4N^2 - \underline{\int} L(x)dx$.
    This algebraic manipulation yields $m(E) \le \underline{\int} L(x)dx \le \overline{\int} L(x)dx \le \overline{\int} U(x)dx \le m(E)$.
    
    The squeeze theorem forces $m(E) = \int L(x)dx = \int U(x)dx$. Since $0 \le L \le U$ and $\int (U-L) = 0$, we must have $U=L$ almost everywhere. This implies $\chi_E(x,y)$ is integrable for almost every $x$, and the iterated integral equals $m(E)$.

    \textbf{Step 2: Proof of the bounding claim.}
    By the definition of Lebesgue outer measure, $m(E) = \inf \displaystyle\sum m(B_j)$ over all countable covers of open boxes $B_j$ such that $E \subseteq \bigcup B_j$. Let $B_j = I_j \times J_j$. The measure of a box trivially respects Fubini: $m(B_j) = m(I_j)m(J_j) = \int_\mathbb{R} \left(\int_\mathbb{R} \chi_{B_j}(x,y)dy\right)dx$.
    
    Let $H(x,y) = \displaystyle\sum \chi_{B_j}(x,y)$ be the sum of indicator functions of the covering boxes. Since $E \subseteq \bigcup B_j$, we have $\chi_E \le H$. Therefore, $U(x) \le \overline{\int} H(x,y)dy$.
    Integrating $U(x)$ over the bounds gives:
    $$ \overline{\int}_{[-N,N]} U(x)dx \le \overline{\int}_{[-N,N]} \int_{[-N,N]} H(x,y)dydx = \int_{[-N,N]} \displaystyle\sum \chi_{I_j}(x) m(J_j) dx $$
    By the Monotone Convergence Theorem for sums, we pull the sum outside:
    $$ = \displaystyle\sum \int \chi_{I_j}(x) m(J_j) dx = \displaystyle\sum m(I_j)m(J_j) = \displaystyle\sum m(B_j) \le m(E) + \epsilon $$
    Since this bound holds for any optimal cover and any $\epsilon > 0$, we conclude that $\overline{\int} U(x)dx \le m(E)$. 
    
    \textbf{Step 3: Conclusion by symmetry.}
    We have established that $\int_{[-N,N]} \left( \int_{[-N,N]} \chi_E(x,y) dy \right) dx = m(E)$. By symmetry, exchanging the roles of $x$ and $y$ in the definitions of the upper and lower integrals yields the exact same bound and squeeze result. Thus, $\int_{[-N,N]} \left( \int_{[-N,N]} \chi_E(x,y) dx \right) dy = m(E)$ as well, completing the proof.
\end{pfbox}


\begin{theorem}[Tonelli]
    Let $f: \mathbb{R}^2 \rightarrow [0, \infty]$ be a non-negative measurable function. Then the slice functions $f(x,y)$ are measurable for almost every $x$ and almost every $y$. The functions $F(x) = \int_\mathbb{R} f(x,y) dy$ and $G(y) = \int_\mathbb{R} f(x,y) dx$ are measurable, and:
    $$ \int_\mathbb{R} \left( \int_\mathbb{R} f(x,y) dy \right) dx = \int_\mathbb{R} \left( \int_\mathbb{R} f(x,y) dx \right) dy = \int_{\mathbb{R}^2} f $$
\end{theorem}

\begin{pfbox}
    \textbf{Step 1: Reduction to simple and characteristic functions on a bounded square.}
    Assume first that $f$ is supported on a bounded square $[-N,N] \times [-N,N]$. Since $f$ is measurable and non-negative, there exists a sequence of simple functions $s_k$ increasing to $f$, with their support also restricted to $[-N,N]^2$.
    
    A simple function is a finite sum $s_k = \displaystyle\sum c_l \chi_{E_l}$. By the preceding Lemma, the theorem holds for the characteristic function of any measurable set $E_l \subseteq [-N,N]^2$ for both integration orders. By linearity of the integral, the theorem also holds for any simple function $s_k$.
    
    Fixing $x$, the sequence $s_k(x, \cdot)$ increases to $f(x, \cdot)$. By the Lebesgue Monotone Convergence Theorem (MCT) applied to the inner integral, $\displaystyle\lim_{k\to\infty} \int_\mathbb{R} s_k(x,y)dy = \int_\mathbb{R} f(x,y)dy$.
    Applying the MCT a second time to the outer integral, we can exchange the limit and integration: 
    $$ \int_\mathbb{R} \left(\lim_{k\to\infty} \int_\mathbb{R} s_k(x,y)dy\right)dx = \lim_{k\to\infty} \int_\mathbb{R} \int_\mathbb{R} s_k(x,y) dy dx = \lim_{k\to\infty} \int_{\mathbb{R}^2} s_k = \int_{\mathbb{R}^2} f $$

    \textbf{Step 2: Reduction to a bounded interval for general non-negative functions.}
    Now let $f \ge 0$ be a general measurable function on $\mathbb{R}^2$. For any integer $N \in \mathbb{N}$, define the localized function $f_N := f \chi_{[-N,N] \times [-N,N]}$.
    These localized functions form a monotonically increasing sequence $0 \le f_1 \le f_2 \dots \le f$ that converges pointwise to $f$. 
    By Step 1, Tonelli's theorem holds for all $f_N$, so $\int_{\mathbb{R}^2} f_N = \int_\mathbb{R} \int_\mathbb{R} f_N(x,y)dy dx$.
    
    By the MCT applied over $\mathbb{R}^2$, $\displaystyle\lim_{N\to\infty} \int_{\mathbb{R}^2} f_N = \int_{\mathbb{R}^2} f$.
    Fixing $x$, the sequence $f_N(x, \cdot)$ increases to $f(x, \cdot)$. Applying the MCT twice exactly as we did in Step 1 allows us to exchange the limit and integration again:
    $$ \int_\mathbb{R} \left(\lim_{N\to\infty} \int_\mathbb{R} f_N(x,y)dy\right)dx = \lim_{N\to\infty} \int_\mathbb{R} \int_\mathbb{R} f_N(x,y) dy dx = \lim_{N\to\infty} \int_{\mathbb{R}^2} f_N = \int_{\mathbb{R}^2} f $$
    
    \textbf{Step 3: Conclusion by symmetry.}
    By symmetry, repeating the identical limiting processes for the reverse integration order ($dx$ inner, $dy$ outer) yields $\int_\mathbb{R} \left( \int_\mathbb{R} f(x,y) dx \right) dy = \int_{\mathbb{R}^2} f$. This completes the proof of Tonelli's Theorem.
\end{pfbox}


\begin{theorem}[Fubini]
    Let $f: \mathbb{R}^2 \rightarrow \mathbb{R}$ be an absolutely integrable function ($f \in L^1(\mathbb{R}^2)$). Then for almost every $x$, $f(x,y)$ is absolutely integrable with respect to $y$, and for almost every $y$, $f(x,y)$ is absolutely integrable with respect to $x$. Furthermore, the functions $F(x) = \int_\mathbb{R} f(x,y) dy$ and $G(y) = \int_\mathbb{R} f(x,y) dx$ are absolutely integrable, and we have:
    $$ \int_\mathbb{R} F(x) dx = \int_\mathbb{R} \left( \int_\mathbb{R} f(x,y) dy \right) dx = \int_\mathbb{R} \left( \int_\mathbb{R} f(x,y) dx \right) dy = \int_\mathbb{R} G(y) dy = \int_{\mathbb{R}^2} f $$
\end{theorem}

\begin{pfbox}
    \textbf{Reduction to non-negative functions.}
    Let $f \in L^1(\mathbb{R}^2)$. We can decompose $f = f^+ - f^-$, where $f^+, f^- \in L^1(\mathbb{R}^2)$ are non-negative measurable functions.
    
    By Tonelli's Theorem (proven above), there exist functions $F^+, F^-: \mathbb{R} \rightarrow [0, \infty]$ such that for almost every $x$, $F^+(x) = \int_\mathbb{R} f^+(x,y)dy$ and $F^-(x) = \int_\mathbb{R} f^-(x,y)dy$. We also know that $\int_\mathbb{R} F^+(x)dx = \int_{\mathbb{R}^2} f^+$ and $\int_\mathbb{R} F^-(x)dx = \int_{\mathbb{R}^2} f^-$.
    
    Since $f \in L^1(\mathbb{R}^2)$, both $\int_{\mathbb{R}^2} f^+$ and $\int_{\mathbb{R}^2} f^-$ are finite. Thus, for almost every $x$, the absolute value integral is finite: 
    $$ \int_\mathbb{R} |f(x,y)|dy = \int_\mathbb{R} f^+(x,y)dy + \int_\mathbb{R} f^-(x,y)dy < \infty $$
    This means $f(x,y)$ is absolutely integrable with respect to $y$ for almost every $x$.
    
    We can safely define $F(x) := \int_\mathbb{R} f(x,y)dy = F^+(x) - F^-(x)$. Since both $F^+$ and $F^-$ are in $L^1(\mathbb{R})$, their difference $F \in L^1(\mathbb{R})$. 
    Integrating $F(x)$ over $\mathbb{R}$ yields:
    $$ \int_\mathbb{R} F(x)dx = \int_\mathbb{R} F^+(x)dx - \int_\mathbb{R} F^-(x)dx = \int_{\mathbb{R}^2} f^+ - \int_{\mathbb{R}^2} f^- = \int_{\mathbb{R}^2} f $$
    
    By symmetry, applying the exact same decomposition $f = f^+ - f^-$ and utilizing Tonelli's theorem for the $dx$ inner integral proves that $G(y) = \int_\mathbb{R} f(x,y)dx \in L^1(\mathbb{R})$, and integrating $G(y)$ with respect to $y$ yields exactly $\int_{\mathbb{R}^2} f$. This establishes Fubini's Theorem in full.
\end{pfbox}
